43static void eth_teardown_locked(void)
44{
45 // Caller must hold eth_lifecycle_mutex
46 // Clear deferred flag and mark shutting down
47 s_deferred_eth_teardown = false;
48 eth_set_shutting_down(true);
49
50 // Stop DHCP client (best effort)
51 if (eth_netif)
52 {
53 esp_err_t r = esp_netif_dhcpc_stop(eth_netif);
54 if (r != ESP_OK && r != ESP_ERR_ESP_NETIF_DHCP_ALREADY_STOPPED)
55 LOGW(TAG, "dhcpc_stop -> %s", esp_err_to_name(r));
56 }
57
58 // Stop driver if running
59 if (eth_handle && eth_started)
60 {
61 esp_err_t r = esp_eth_stop(eth_handle);
62 if (r != ESP_OK)
63 LOGW(TAG, "esp_eth_stop -> %s", esp_err_to_name(r));
64 eth_started = false;
65 }
66
67 // Defensive: remove any GPIO ISR for W5500 interrupt pin (if used)
68 if (hw.w5500_int_gpio >= 0)
69 {
70 esp_err_t r = gpio_isr_handler_remove((gpio_num_t)hw.w5500_int_gpio);
71 if (r != ESP_OK && r != ESP_ERR_INVALID_ARG)
72 LOGW(TAG, "gpio_isr_handler_remove(w5500_int) -> %s", esp_err_to_name(r));
73 }
74
75 // Delete glue if present
76 if (eth_glue)
77 {
78 esp_err_t r = esp_eth_del_netif_glue(eth_glue);
79 if (r != ESP_OK)
80 LOGW(TAG, "esp_eth_del_netif_glue -> %s", esp_err_to_name(r));
81 eth_glue = NULL;
82 }
83
84 // Destroy the LWIP ESP netif before recreating it. Leaving a stale ETH_DEF
85 // netif attached is what causes the classic "netif already added" assert when
86 // the driver is re-initialized after unplug/reconnect.
87 if (eth_netif)
88 {
89 esp_netif_destroy(eth_netif);
90 eth_netif = NULL;
91 }
92
93 if (eth_handle)
94 {
95 esp_err_t r = esp_eth_driver_uninstall(eth_handle);
96 if (r != ESP_OK)
97 LOGW(TAG, "esp_eth_driver_uninstall -> %s", esp_err_to_name(r));
98 eth_handle = NULL;
99 }
100
101 s_eth_initialized = false;
102 eth_started = false;
103
104 LOGI(TAG, "eth_teardown_locked: complete");
105
106 // Clear shutting down flag (caller will release mutex)
107 eth_set_shutting_down(false);
108}

145esp_err_t eth_hardware_reset(void)
146{
147 // Ensure lifecycle mutex exists
148 eth_lifecycle_init_once();
149
150 // Try to take lifecycle mutex; if busy, defer and return busy
151 if (xSemaphoreTake(eth_lifecycle_mutex, pdMS_TO_TICKS(500)) != pdTRUE)
152 {
153 LOGW(TAG, "eth_hardware_reset: lifecycle mutex busy; cannot perform hardware reset now");
154 return ESP_ERR_TIMEOUT;
155 }
156
157 LOGI(TAG, "eth_hardware_reset: starting guarded hardware reset");
158 LOGI(TAG, "eth_hardware_reset: DMA free=%u internal free=%u",
159 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
160
161 // Mark shutting down to prevent event handlers from racing
162 eth_set_shutting_down(true);
163
164 // Stop DHCP client (best effort)
165 if (eth_netif)
166 {
167 esp_err_t r = esp_netif_dhcpc_stop(eth_netif);
168 if (r != ESP_OK && r != ESP_ERR_ESP_NETIF_DHCP_ALREADY_STOPPED)
169 LOGW(TAG, "eth_hardware_reset: dhcpc_stop -> %s", esp_err_to_name(r));
170 }
171
172 // Stop driver if running
173 if (eth_handle && eth_started)
174 {
175 esp_err_t r = esp_eth_stop(eth_handle);
176 if (r != ESP_OK)
177 LOGW(TAG, "eth_hardware_reset: esp_eth_stop -> %s", esp_err_to_name(r));
178 eth_started = false;
179 }
180
181 // Optionally toggle reset GPIO if configured
182 if (hw.w5500_rst_gpio >= 0)
183 {
184 gpio_config_t io_conf = {
185 .mode = GPIO_MODE_OUTPUT,
186 .pin_bit_mask = (1ULL << hw.w5500_rst_gpio),
187 .pull_down_en = 0,
188 .pull_up_en = 0,
189 .intr_type = GPIO_INTR_DISABLE};
190 gpio_config(&io_conf);
191
192 LOGI(TAG, "eth_hardware_reset: asserting W5500 reset GPIO %d", hw.w5500_rst_gpio);
193 gpio_set_level((gpio_num_t)hw.w5500_rst_gpio, 0); // assert reset (active low)
194 vTaskDelay(pdMS_TO_TICKS(200)); // hold reset long enough for W5500 to enter reset reliably
195
196 LOGI(TAG, "eth_hardware_reset: releasing W5500 reset GPIO %d", hw.w5500_rst_gpio);
197 gpio_set_level((gpio_num_t)hw.w5500_rst_gpio, 1); // release reset
198 vTaskDelay(pdMS_TO_TICKS(500)); // allow W5500 to boot and settle before re-start
199 }
200 else
201 {
202 LOGW(TAG, "eth_hardware_reset: no reset GPIO configured; performing software reinit only");
203 vTaskDelay(pdMS_TO_TICKS(50));
204 }
205
206 // Reinitialize driver and netif if needed.
207 if (!s_eth_initialized || eth_handle == NULL || eth_netif == NULL)
208 {
209 LOGI(TAG, "eth_hardware_reset: calling eth_init()");
210 LOGI(TAG, "eth_hardware_reset: before eth_init DMA free=%u internal free=%u",
211 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
212
213 // Release lifecycle mutex before calling eth_init() to avoid deadlock
214 xSemaphoreGive(eth_lifecycle_mutex);
215 eth_init();
216
217 // Re-acquire lifecycle mutex to continue; handle failure
218 if (xSemaphoreTake(eth_lifecycle_mutex, pdMS_TO_TICKS(200)) != pdTRUE)
219 {
220 LOGW(TAG, "eth_hardware_reset: failed to re-take lifecycle mutex after eth_init");
221 eth_set_shutting_down(false);
222 return ESP_ERR_TIMEOUT;
223 }
224 }
225
226 // Start driver if not started
227 if (!eth_started && eth_handle != NULL)
228 {
229 LOGI(TAG, "eth_hardware_reset: before esp_eth_start DMA free=%u internal free=%u",
230 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
231 esp_err_t r = esp_eth_start(eth_handle);
232 if (r != ESP_OK)
233 {
234 LOGW(TAG, "eth_hardware_reset: esp_eth_start -> %s", esp_err_to_name(r));
235 }
236 else
237 {
238 eth_started = true;
239 LOGI(TAG, "eth_hardware_reset: esp_eth_start succeeded");
240 LOGI(TAG, "eth_hardware_reset: after esp_eth_start DMA free=%u internal free=%u",
241 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
242 }
243 }
244
245 // Clear shutting down flag and release mutex
246 eth_set_shutting_down(false);
247 xSemaphoreGive(eth_lifecycle_mutex);
248
249 LOGI(TAG, "eth_hardware_reset: complete");
250 return ESP_OK;
251}

262esp_err_t eth_hardware_reset_safe(void)
263{
264 esp_err_t rc;
265
266 // 1) Try software restart first (fast, non-invasive)
267 LOGI(TAG, "eth_hardware_reset_safe: attempting software restart first");
268 rc = eth_software_restart();
269 if (rc == ESP_OK)
270 {
271 LOGI(TAG, "eth_hardware_reset_safe: eth_software_restart succeeded; allow settle and verify link");
272
273 // Allow a short settle period and verify link/IP
274 vTaskDelay(pdMS_TO_TICKS(500));
275
276 // If link is up and IP present, consider success
277 if (ethernet_link_is_up())
278 {
279 LOGI(TAG, "eth_hardware_reset_safe: link is up after software restart");
280 return ESP_OK;
281 }
282
283 LOGW(TAG, "eth_hardware_reset_safe: link not up after software restart; will attempt hardware reset");
284 }
285 else
286 {
287 LOGW(TAG, "eth_hardware_reset_safe: eth_software_restart returned %s; will attempt hardware reset", esp_err_to_name(rc));
288 }
289
290 // 2) Fall back to guarded hardware reset (this function already handles lifecycle mutex)
291 LOGI(TAG, "eth_hardware_reset_safe: attempting guarded hardware reset");
292 rc = eth_hardware_reset();
293 if (rc == ESP_OK)
294 {
295 // Allow PHY to boot and DHCP to start
296 vTaskDelay(pdMS_TO_TICKS(500));
297 LOGI(TAG, "eth_hardware_reset_safe: guarded hardware reset succeeded");
298 return ESP_OK;
299 }
300
301 LOGE(TAG, "eth_hardware_reset_safe: guarded hardware reset failed: %s", esp_err_to_name(rc));
302 return rc;
303}

328void eth_teardown_safe(void)
329{
330 eth_lifecycle_init_once();
331
332 // Try to take lifecycle mutex briefly; if busy, mark deferred and return
333 if (xSemaphoreTake(eth_lifecycle_mutex, pdMS_TO_TICKS(200)) != pdTRUE)
334 {
335 LOGW(TAG, "eth_teardown_safe: lifecycle mutex busy; deferring teardown");
336 s_deferred_eth_teardown = true;
337 return;
338 }
339
340 // We hold the mutex now; perform locked teardown
341 eth_teardown_locked();
342
343 // Release mutex
344 xSemaphoreGive(eth_lifecycle_mutex);
345}

382void eth_init(void)
383{
384 if (!hw.has_ethernet)
385 {
386 LOGI(TAG, "eth_init: Ethernet not present on this hardware");
387 return;
388 }
389
390 if (eth_lifecycle_mutex == NULL)
391 eth_lifecycle_init_once();
392
393 if (xSemaphoreTake(eth_lifecycle_mutex, pdMS_TO_TICKS(5000)) != pdTRUE)
394 {
395 LOGE(TAG, "eth_init: failed to take lifecycle mutex");
396 return;
397 }
398
399 if (s_eth_initialized && eth_handle != NULL)
400 {
401 LOGD(TAG, "eth_init: already initialized");
402 xSemaphoreGive(eth_lifecycle_mutex);
403 return;
404 }
405
406 LOGI(TAG, "eth_init: initializing W5500 Ethernet (safe)");
407
408 LOGI(TAG, "eth_init: entry DMA free=%u internal free=%u", (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
409
410 esp_err_t err = esp_netif_init();
411 if (err != ESP_OK && err != ESP_ERR_INVALID_STATE)
412 {
413 LOGE(TAG, "eth_init: esp_netif_init failed: %s", esp_err_to_name(err));
414 xSemaphoreGive(eth_lifecycle_mutex);
415 return;
416 }
417
418 // Always create a fresh ETH_DEF netif after teardown. Reusing a stale
419 // handle here is what causes the lwIP assert: "netif already added".
420 esp_netif_t *existing = esp_netif_get_handle_from_ifkey("ETH_DEF");
421 if (existing != NULL)
422 {
423 LOGW(TAG, "eth_init: stale ETH_DEF netif found (%p); destroying before recreating", (void *)existing);
424 esp_netif_destroy(existing);
425 }
426
427 esp_netif_config_t cfg = ESP_NETIF_DEFAULT_ETH();
428 eth_netif = esp_netif_new(&cfg);
429 if (eth_netif == NULL)
430 {
431 LOGE(TAG, "eth_init: esp_netif_new returned NULL");
432 xSemaphoreGive(eth_lifecycle_mutex);
433 return;
434 }
435 LOGI(TAG, "eth_init: created new ETH_DEF netif %p", (void *)eth_netif);
436
437 // SPI bus init (idempotent)
438 spi_bus_config_t buscfg = {
439 .miso_io_num = hw.w5500_miso_gpio,
440 .mosi_io_num = hw.w5500_mosi_gpio,
441 .sclk_io_num = hw.w5500_sclk_gpio,
442 .quadwp_io_num = -1,
443 .quadhd_io_num = -1,
444 .max_transfer_sz = 1024,
445 };
446
447 // Keep DMA enabled so the W5500 driver can read/write larger SPI bursts.
448 // Lower the queue depth to reduce outstanding DMA-backed transactions.
449 err = spi_bus_initialize(W5500_SPI_HOST, &buscfg, SPI_DMA_CH_AUTO);
450 if (err != ESP_OK && err != ESP_ERR_INVALID_STATE)
451 {
452 LOGE(TAG, "eth_init: spi_bus_initialize failed: %s", esp_err_to_name(err));
453 xSemaphoreGive(eth_lifecycle_mutex);
454 return;
455 }
456
457 LOGI(TAG, "eth_init: after spi_bus_initialize DMA free=%u internal free=%u",
458 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
459
460 // SPI device config
461 static spi_device_interface_config_t w5500_spi_devcfg = {
462 .mode = 0,
463 .clock_speed_hz = 20 * 1000 * 1000,
464 .spics_io_num = -1,
465 .queue_size = 2, // conservative: reduce outstanding DMA-backed transactions
466 };
467 w5500_spi_devcfg.spics_io_num = hw.w5500_cs_gpio;
468
469 // MAC + PHY
470 eth_mac_config_t mac_config = ETH_MAC_DEFAULT_CONFIG();
471 eth_phy_config_t phy_config = ETH_PHY_DEFAULT_CONFIG();
472 if (hw.w5500_rst_gpio >= 0)
473 phy_config.reset_gpio_num = hw.w5500_rst_gpio;
474 phy_config.phy_addr = 0;
475
476 eth_w5500_config_t w5500_cfg = {0};
477 w5500_cfg.spi_host_id = W5500_SPI_HOST;
478 w5500_cfg.spi_devcfg = &w5500_spi_devcfg;
479 if (hw.w5500_int_gpio >= 0)
480 w5500_cfg.int_gpio_num = hw.w5500_int_gpio;
481
482 esp_eth_mac_t *mac = esp_eth_mac_new_w5500(&w5500_cfg, &mac_config);
483 if (mac == NULL)
484 {
485 LOGE(TAG, "eth_init: esp_eth_mac_new_w5500 returned NULL");
486 xSemaphoreGive(eth_lifecycle_mutex);
487 return;
488 }
489
490 esp_eth_phy_t *phy = esp_eth_phy_new_w5500(&phy_config);
491 if (phy == NULL)
492 {
493 LOGE(TAG, "eth_init: esp_eth_phy_new_w5500 returned NULL");
494 xSemaphoreGive(eth_lifecycle_mutex);
495 return;
496 }
497
498 // Install driver with default config (MAC and PHY are already configured)
499 esp_eth_config_t eth_config = ETH_DEFAULT_CONFIG(mac, phy);
500 err = esp_eth_driver_install(&eth_config, &eth_handle);
501 if (err != ESP_OK)
502 {
503 LOGE(TAG, "eth_init: esp_eth_driver_install failed: %s", esp_err_to_name(err));
504 eth_handle = NULL;
505 xSemaphoreGive(eth_lifecycle_mutex);
506 return;
507 }
508
509 uint8_t mac_addr[6] = {0};
510 err = esp_read_mac(mac_addr, ESP_MAC_ETH);
511 if (err != ESP_OK)
512 {
513 LOGW(TAG, "eth_init: esp_read_mac(ESP_MAC_ETH) failed: %s; falling back to default MAC", esp_err_to_name(err));
514 err = esp_efuse_mac_get_default(mac_addr);
515 }
516
517 if (err == ESP_OK)
518 {
519 mac_addr[0] &= 0xFE; // clear multicast bit
520 mac_addr[0] |= 0x02; // set locally administered bit
521 err = esp_eth_ioctl(eth_handle, ETH_CMD_S_MAC_ADDR, mac_addr);
522 if (err != ESP_OK)
523 {
524 LOGW(TAG, "eth_init: failed to set W5500 MAC address: %s", esp_err_to_name(err));
525 }
526 else
527 {
528 uint8_t programmed_mac[6] = {0};
529 if (esp_eth_ioctl(eth_handle, ETH_CMD_G_MAC_ADDR, programmed_mac) == ESP_OK)
530 {
531 LOGI(TAG, "eth_init: W5500 MAC programmed as %02X:%02X:%02X:%02X:%02X:%02X",
532 programmed_mac[0], programmed_mac[1], programmed_mac[2], programmed_mac[3], programmed_mac[4], programmed_mac[5]);
533 }
534 else
535 {
536 LOGI(TAG, "eth_init: set W5500 MAC to %02X:%02X:%02X:%02X:%02X:%02X",
537 mac_addr[0], mac_addr[1], mac_addr[2], mac_addr[3], mac_addr[4], mac_addr[5]);
538 }
539 }
540 }
541 else
542 {
543 LOGW(TAG, "eth_init: could not obtain a valid MAC address for the W5500");
544 }
545
546 LOGI(TAG, "eth_init: after esp_eth_driver_install DMA free=%u internal free=%u",
547 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
548
549 // Attach netif glue
550 eth_glue = esp_eth_new_netif_glue(eth_handle);
551 if (eth_glue == NULL)
552 {
553 LOGE(TAG, "eth_init: eth_glue NULL, aborting");
554 if (eth_handle)
555 {
556 esp_eth_driver_uninstall(eth_handle);
557 eth_handle = NULL;
558 }
559 xSemaphoreGive(eth_lifecycle_mutex);
560 return;
561 }
562
563 // Defensive guard: do not attach a netif that is already linked to lwIP.
564 if (eth_netif == NULL)
565 {
566 LOGE(TAG, "eth_init: ETH_DEF netif vanished before attach");
567 esp_eth_del_netif_glue(eth_glue);
568 eth_glue = NULL;
569 esp_eth_driver_uninstall(eth_handle);
570 eth_handle = NULL;
571 xSemaphoreGive(eth_lifecycle_mutex);
572 return;
573 }
574
575 err = esp_netif_attach(eth_netif, eth_glue);
576
577 if (err != ESP_OK)
578 {
579 LOGE(TAG, "esp_netif_attach failed: %s", esp_err_to_name(err));
580 esp_eth_del_netif_glue(eth_glue);
581 eth_glue = NULL;
582 esp_eth_driver_uninstall(eth_handle);
583 eth_handle = NULL;
584 xSemaphoreGive(eth_lifecycle_mutex);
585 return;
586 }
587
588 LOGI(TAG, "eth_init: W5500 initialized and attached to netif");
589 s_eth_initialized = true;
590 eth_started = false;
591
592 // Start DHCP client if configured for DHCP
593 if (ethernet_config_is_dhcp_enabled())
594 {
595 esp_err_t d = esp_netif_dhcpc_start(eth_netif);
596 if (d != ESP_OK && d != ESP_ERR_ESP_NETIF_DHCP_ALREADY_STARTED)
597 LOGW(TAG, "eth_init: esp_netif_dhcpc_start failed: %s", esp_err_to_name(d));
598 else
599 LOGI(TAG, "eth_init: DHCP client started on ETH_DEF");
600 }
601
602 LOGI(TAG, "eth_init: exit DMA free=%u internal free=%u",
603 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
604
605 xSemaphoreGive(eth_lifecycle_mutex);
606}

617bool ethernet_link_is_up(void)
618{
619 if (eth_is_shutting_down())
620 return false;
621
622#if defined(ETH_CMD_G_PHY_LINK)
623 if (eth_handle != NULL)
624 {
625 int link = 0;
626 esp_err_t r = esp_eth_ioctl(eth_handle, ETH_CMD_G_PHY_LINK, &link);
627 if (r == ESP_OK)
628 {
629 return (link != 0);
630 }
631 // fall through to netif/IP fallback on ioctl failure
632 }
633#endif
634
635 if (eth_handle == NULL || eth_netif == NULL)
636 return false;
637
638 esp_netif_ip_info_t ip;
639 if (esp_netif_get_ip_info(eth_netif, &ip) != ESP_OK)
640 return false;
641
642 return ip.ip.addr != 0;
643}

651void attempt_guarded_eth_restart(void)
652{
653 // If link/netif appears up, skip restart (DHCP may still be negotiating)
654 if (ethernet_link_is_up())
655 {
656 LOGI(TAG, "attempt_guarded_eth_restart: link/netif up; skipping restart");
657 s_eth_restart_attempts = 0;
658 return;
659 }
660
661 // Backoff limit to avoid restart storms
662 if (s_eth_restart_attempts >= ETH_RESTART_MAX_ATTEMPTS)
663 {
664 LOGW(TAG, "attempt_guarded_eth_restart: max attempts reached; delaying longer");
665 vTaskDelay(pdMS_TO_TICKS(60000));
666 s_eth_restart_attempts = 0;
667 return;
668 }
669
670 s_eth_restart_attempts++;
671 LOGI(TAG, "attempt_guarded_eth_restart: attempt %d/%d", s_eth_restart_attempts, ETH_RESTART_MAX_ATTEMPTS);
672
673 // Try a software restart first via the safe wrapper (which will fall back to hardware reset)
674 esp_err_t rc = eth_hardware_reset_safe();
675 if (rc == ESP_OK)
676 {
677 LOGI(TAG, "attempt_guarded_eth_restart: eth_hardware_reset_safe succeeded");
678
679 // Request the network manager to refresh sm_actuals
680 network_manager_request_actuals_update();
681
682 // Request the syslog task to update its state (non-blocking)
683 update_syslog_state();
684
685 // Allow link/IP to settle a bit
686 vTaskDelay(ETH_RESTART_BACKOFF_TICKS);
687 return;
688 }
689
690 LOGW(TAG, "attempt_guarded_eth_restart: eth_hardware_reset_safe failed (%s); attempting full teardown/reinit as fallback", esp_err_to_name(rc));
691
692 // If safe reset failed, attempt a full teardown (deferred if lifecycle busy)
693 eth_teardown_safe();
694 if (s_deferred_eth_teardown)
695 {
696 LOGW(TAG, "attempt_guarded_eth_restart: teardown deferred; will retry later");
697 return;
698 }
699
700 // Small settle delay before reinit to allow any pending events to clear and avoid tight restart loops
701 vTaskDelay(pdMS_TO_TICKS(200));
702
703 // Reinitialize and start
704 eth_init();
705 vTaskDelay(pdMS_TO_TICKS(200));
706 eth_start();
707
708 // Allow link/IP to settle a bit before any further attempts
709 vTaskDelay(ETH_RESTART_BACKOFF_TICKS);
710}

718void handle_deferred_eth_teardown_if_needed(void)
719{
720 if (!s_deferred_eth_teardown)
721 return;
722
723 LOGI(TAG, "handle_deferred_eth_teardown_if_needed: attempting deferred teardown");
724
725 if (eth_lifecycle_mutex == NULL)
726 eth_lifecycle_init_once();
727
728 if (xSemaphoreTake(eth_lifecycle_mutex, pdMS_TO_TICKS(200)) != pdTRUE)
729 {
730 LOGW(TAG, "handle_deferred_eth_teardown_if_needed: lifecycle mutex still busy");
731 return;
732 }
733
734 // Clear deferred flag and perform locked teardown
735 s_deferred_eth_teardown = false;
736 eth_teardown_locked();
737
738 LOGI(TAG, "handle_deferred_eth_teardown_if_needed: deferred teardown complete");
739
740 xSemaphoreGive(eth_lifecycle_mutex);
741}

749void eth_start(void)
750{
751 if (!hw.has_ethernet)
752 {
753 LOGI(TAG, "eth_start: Ethernet not present");
754 return;
755 }
756
757 if (eth_lifecycle_mutex == NULL)
758 eth_lifecycle_init_once();
759
760 if (xSemaphoreTake(eth_lifecycle_mutex, pdMS_TO_TICKS(5000)) != pdTRUE)
761 {
762 LOGE(TAG, "eth_start: failed to take lifecycle mutex");
763 return;
764 }
765
766 if (!s_eth_initialized || eth_handle == NULL)
767 {
768 LOGW(TAG, "eth_start: not initialized; calling eth_init()");
769 eth_init();
770 // eth_init will release the mutex; reacquire to continue
771 if (xSemaphoreTake(eth_lifecycle_mutex, pdMS_TO_TICKS(5000)) != pdTRUE)
772 {
773 LOGE(TAG, "eth_start: failed to re-take lifecycle mutex after init");
774 return;
775 }
776 }
777
778 if (eth_started)
779 {
780 LOGD(TAG, "eth_start: already started");
781 xSemaphoreGive(eth_lifecycle_mutex);
782 return;
783 }
784
785 if (eth_handle == NULL)
786 {
787 LOGW(TAG, "eth_start: eth_handle NULL, cannot start");
788 xSemaphoreGive(eth_lifecycle_mutex);
789 return;
790 }
791
792 LOGI(TAG, "eth_start: before esp_eth_start DMA free=%u internal free=%u",
793 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
794
795 esp_err_t err = esp_eth_start(eth_handle);
796
797 if (err != ESP_OK)
798 {
799 LOGE(TAG, "eth_start: esp_eth_start failed: %s", esp_err_to_name(err));
800 LOGW(TAG, "eth_start: retrying once after a brief stop/start cycle");
801
802 esp_err_t stop_err = esp_eth_stop(eth_handle);
803 if (stop_err != ESP_OK && stop_err != ESP_ERR_INVALID_STATE)
804 {
805 LOGW(TAG, "eth_start: esp_eth_stop during retry -> %s", esp_err_to_name(stop_err));
806 }
807
808 vTaskDelay(pdMS_TO_TICKS(500));
809 err = esp_eth_start(eth_handle);
810 }
811
812 if (err != ESP_OK)
813 {
814 LOGE(TAG, "eth_start: retry start failed: %s", esp_err_to_name(err));
815 }
816 else
817 {
818 eth_started = true;
819 LOGI(TAG, "eth_start: Ethernet driver started");
820 LOGI(TAG, "eth_start: after esp_eth_start DMA free=%u internal free=%u",
821 (unsigned)heap_caps_get_free_size(MALLOC_CAP_DMA), (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL));
822 }
823
824 xSemaphoreGive(eth_lifecycle_mutex);
825}

976esp_err_t ethernet_config_apply(void)
977{
978 char hostname[32] = {0};
979
980 if (!eth_netif && hw.has_ethernet)
981 {
982 LOGW(TAG, "eth_netif is NULL; call eth_init() before applying config");
983 return ESP_ERR_INVALID_STATE;
984 }
985
986 Ethernet_config_t local;
987 if (network_config_mutex && xSemaphoreTake(network_config_mutex, pdMS_TO_TICKS(100)) == pdTRUE)
988 {
989 LOGD(TAG, "enable_ethernet: %s, enable_dhcp: %s", ethernet_config.enable_ethernet, ethernet_config.enable_dhcp);
990 memcpy(&local, &ethernet_config, sizeof(Ethernet_config_t));
991 strncpy(hostname, wifi_config.hostname, sizeof(hostname) - 1);
992 xSemaphoreGive(network_config_mutex);
993 }
994 else
995 {
996 LOGW(TAG, "Could not take network_config_mutex; aborting ethernet_config_apply");
997 return ESP_ERR_TIMEOUT;
998 }
999
1000 if (strcmp(local.enable_ethernet, "On") != 0)
1001 {
1002 LOGI(TAG, "Ethernet disabled in config; stopping driver");
1003 eth_stop();
1004 return ESP_OK;
1005 }
1006
1007 LOGD(TAG, "Setting Ethernet hostname to: %s", hostname);
1008 esp_err_t herr = esp_netif_set_hostname(eth_netif, hostname);
1009 if (herr != ESP_OK)
1010 {
1011 LOGW(TAG, "Failed to set Ethernet hostname: %s", esp_err_to_name(herr));
1012 }
1013
1014 LOGD(TAG, "Ethernet enabled in config; starting driver if needed");
1015 eth_start();
1016
1017 if (strcmp(local.enable_dhcp, "On") == 0)
1018 {
1019 LOGD(TAG, "Starting DHCP client on Ethernet");
1020 esp_err_t err = esp_netif_dhcpc_start(eth_netif);
1021 if (err != ESP_OK && err != ESP_ERR_ESP_NETIF_DHCP_ALREADY_STARTED)
1022 {
1023 LOGE(TAG, "esp_netif_dhcpc_start failed: %s", esp_err_to_name(err));
1024 return err;
1025 }
1026 return ESP_OK;
1027 }
1028
1029 if (!validate_ipv4_string(local.ip_address) ||
1030 !validate_ipv4_string(local.subnet_mask) ||
1031 !validate_ipv4_string(local.gateway) ||
1032 !validate_ipv4_string(local.primary_dns))
1033 {
1034 LOGE(TAG, "Invalid static IP configuration; not applying");
1035 return ESP_ERR_INVALID_ARG;
1036 }
1037
1038 esp_netif_ip_info_t ip_info;
1039 memset(&ip_info, 0, sizeof(ip_info));
1040 if (inet_pton(AF_INET, local.ip_address, &ip_info.ip) != 1 ||
1041 inet_pton(AF_INET, local.subnet_mask, &ip_info.netmask) != 1 ||
1042 inet_pton(AF_INET, local.gateway, &ip_info.gw) != 1)
1043 {
1044 LOGE(TAG, "inet_pton failed for static IP fields");
1045 return ESP_ERR_INVALID_ARG;
1046 }
1047
1048 if (esp_netif_set_ip_info(eth_netif, &ip_info) != ESP_OK)
1049 {
1050 LOGE(TAG, "esp_netif_set_ip_info failed");
1051 return ESP_FAIL;
1052 }
1053
1054 esp_netif_dns_info_t dns;
1055 memset(&dns, 0, sizeof(dns));
1056 if (inet_pton(AF_INET, local.primary_dns, &dns.ip.u_addr.ip4) == 1)
1057 {
1058 dns.ip.type = ESP_IPADDR_TYPE_V4;
1059 esp_netif_set_dns_info(eth_netif, ESP_NETIF_DNS_MAIN, &dns);
1060 }
1061 else
1062 {
1063 LOGW(TAG, "Primary DNS invalid; skipping");
1064 }
1065
1066 if (strlen(local.secondary_dns) > 0)
1067 {
1068 memset(&dns, 0, sizeof(dns));
1069 if (inet_pton(AF_INET, local.secondary_dns, &dns.ip.u_addr.ip4) == 1)
1070 {
1071 dns.ip.type = ESP_IPADDR_TYPE_V4;
1072 esp_netif_set_dns_info(eth_netif, ESP_NETIF_DNS_BACKUP, &dns);
1073 }
1074 else
1075 {
1076 LOGW(TAG, "Secondary DNS invalid; skipping");
1077 }
1078 }
1079
1080 LOGI(TAG, "Static IP applied: %s", local.ip_address);
1081 return ESP_OK;
1082}
