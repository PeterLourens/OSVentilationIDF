25/* ISR: enqueue a simple event (timestamp not needed) */

36void pushbutton_init(void)

43 .pull_up_en = GPIO_PULLUP_ENABLE,

45 .intr_type = GPIO_INTR_NEGEDGE};

48 if (s_pushbutton_evt_queue == NULL)

55 // Do not re-install ISR service if app_main already did it.

58 if (r != ESP_OK && r != ESP_ERR_INVALID_STATE)

60 ESP_LOGW(TAG, "gpio_install_isr_service failed: %s", esp_err_to_name(r));

64
