193static esp_err_t websocket_handler(httpd_req_t *req)
194{
195 httpd_ws_frame_t ws_pkt;
196 uint8_t *buf = NULL;
197
198 memset(&ws_pkt, 0, sizeof(httpd_ws_frame_t));
199
200 // LOGD(TAG, "WS payload received: '%s' (req=%p)", (char *)ws_pkt.payload, (void *)req);
201
202 if (req->method == HTTP_GET)
203 {
204 LOGD(TAG, "Handshake done, the new connection was opened");
205 http_server = req->handle;
206 net_mgr_notify_eth_activity(); // count handshake as activity
207 return ESP_OK;
208 }
209
210 ws_pkt.type = HTTPD_WS_TYPE_TEXT;
211
212 // Set max_len = 0 to get the frame len
213 esp_err_t ret = httpd_ws_recv_frame(req, &ws_pkt, 0);
214
215 if (ret != ESP_OK)
216 {
217 LOGW(TAG, "httpd_ws_recv_frame failed to get frame len with %d", ret);
218 return ret;
219 }
220
221 if (ws_pkt.len)
222 {
223 buf = heap_caps_calloc(1, ws_pkt.len + 1, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
224 if (buf == NULL)
225 {
226 LOGE(TAG, "Failed to calloc memory for buf");
227 return ESP_ERR_NO_MEM;
228 }
229
230 ws_pkt.payload = buf;
231
232 // Set max_len = ws_pkt.len to get the frame payload
233 ret = httpd_ws_recv_frame(req, &ws_pkt, ws_pkt.len);
234 if (ret != ESP_OK)
235 {
236 LOGW(TAG, "httpd_ws_recv_frame failed with %d", ret);
237 free(buf);
238 return ret;
239 }
240
241 if (strcmp((char *)ws_pkt.payload, "Start") == 0)
242 {
243 esp_log_set_vprintf(websocket_log_output);
244 LOGD(TAG, "Redirecting console to websocket");
245 }
246
247 if (strcmp((char *)ws_pkt.payload, "Stop") == 0)
248 {
249 LOGD(TAG, "Stopped redirecting console to websocket");
250 esp_log_set_vprintf(uart_log_vprintf);
251 }
252
253 if (strcmp((char *)ws_pkt.payload, "stats") == 0)
254 {
255 send_stats_json_ws(req);
256 }
257
258 if (strcmp((char *)ws_pkt.payload, "index") == 0)
259 {
260 LOGD(TAG, "Index data requested via WebSocket");
261 send_index_html_ws(req);
262 }
263
264 if (strcmp((char *)ws_pkt.payload, "sensor_config") == 0)
265 {
266 LOGD(TAG, "Sensor config requested via WebSocket");
267 send_sensor_config_json_ws(req);
268 }
269
270 if (strcmp((char *)ws_pkt.payload, "config") == 0)
271 {
272 LOGD(TAG, "Config requested via WebSocket");
273 send_config_json_ws(req);
274 }
275
276 if (strcmp((char *)ws_pkt.payload, "statemachine") == 0)
277 {
278 LOGD(TAG, "Statemachine config requested via WebSocket");
279 send_statemachine_config_json_ws(req);
280 }
281 free(buf);
282 }
283
284 return ESP_OK;
285}

290static esp_err_t sensorconfig_bus0_post_handler(httpd_req_t *req)
291{
292 size_t len = req->content_len;
293 if (len == 0)
294 return httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "empty body");
295
296 char *buf = heap_caps_malloc(len + 1, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
297 if (!buf)
298 return httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "mem");
299
300 size_t rec = 0;
301 while (rec < len)
302 {
303 int r = httpd_req_recv(req, buf + rec, len - rec);
304 if (r <= 0)
305 {
306 free(buf);
307 return httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "recv");
308 }
309 rec += (size_t)r;
310 }
311 buf[len] = '\0';
312
313 cJSON *arr = build_bus_config_from_body_urlencoded(buf, len, "b0_s");
314 free(buf);
315 if (!arr)
316 return httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "parse fail");
317
318 const char *path = "/data/json/bus0_sensor_config.json";
319 if (!write_json_atomic_fs(path, arr))
320 {
321 cJSON_Delete(arr);
322 return httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "write fail");
323 }
324 cJSON_Delete(arr);
325
326 // reload into runtime globals (uses mutex inside load_sensor_config)
327 if (load_sensor_config("bus0_sensor_config") != ESP_OK)
328 {
329 LOGE(TAG, "Failed to reload bus0 config after write");
330 }
331
332 // After reconfig of sensors the sensors should re-init
333 reload_and_init_sensors();
334
335 // return the page body (original sensor_config.html)
336 httpd_resp_set_type(req, "text/html");
337 len = sensor_config_html_end - sensor_config_html_start;
338 net_mgr_notify_eth_activity();
339 return httpd_resp_send(req, sensor_config_html_start, len);
340}

1203static esp_err_t handle_get_all_configs_tar(httpd_req_t *req)
1204{
1205 DIR *dir = opendir("/data/json");
1206 if (!dir)
1207 {
1208 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "Cannot open config directory");
1209 return ESP_FAIL;
1210 }
1211
1212 httpd_resp_set_type(req, "application/x-tar");
1213 httpd_resp_set_hdr(req, "Content-Disposition", "attachment; filename=\"configs.tar\"");
1214
1215 struct dirent *entry;
1216 uint8_t buf[512];
1217 char path[300];
1218
1219 while ((entry = readdir(dir)) != NULL)
1220 {
1221 if (!strstr(entry->d_name, ".json"))
1222 continue;
1223
1224 snprintf(path, sizeof(path), "/data/json/%s", entry->d_name);
1225
1226 FILE *f = fopen(path, "r");
1227 if (!f)
1228 continue;
1229
1230 fseek(f, 0, SEEK_END);
1231 long size = ftell(f);
1232 fseek(f, 0, SEEK_SET);
1233
1234 // build tar header locally and send it raw
1235 tar_header_t hdr;
1236 memset(&hdr, 0, sizeof(hdr));
1237 strncpy(hdr.name, entry->d_name, sizeof(hdr.name) - 1);
1238 hdr.name[sizeof(hdr.name) - 1] = '\0';
1239 snprintf(hdr.mode, sizeof(hdr.mode), "%07o", 0644);
1240 snprintf(hdr.uid, sizeof(hdr.uid), "%07o", 0);
1241 snprintf(hdr.gid, sizeof(hdr.gid), "%07o", 0);
1242 snprintf(hdr.size, sizeof(hdr.size), "%011o", (unsigned int)size);
1243 snprintf(hdr.mtime, sizeof(hdr.mtime), "%011o", (unsigned int)time(NULL));
1244 memcpy(hdr.magic, "ustar", 5);
1245 memcpy(hdr.version, "00", 2);
1246 hdr.typeflag = '0';
1247 tar_fill_checksum(&hdr);
1248
1249 httpd_resp_send_chunk(req, (const char *)&hdr, 512);
1250
1251 long remaining = size;
1252 while (remaining > 0)
1253 {
1254 size_t chunk = remaining > sizeof(buf) ? sizeof(buf) : remaining;
1255 size_t r = fread(buf, 1, chunk, f);
1256 if (r == 0)
1257 break;
1258 httpd_resp_send_chunk(req, (const char *)buf, r);
1259 remaining -= r;
1260 }
1261
1262 fclose(f);
1263
1264 size_t pad = (512 - (size % 512)) % 512;
1265 if (pad > 0)
1266 {
1267 memset(buf, 0, pad);
1268 httpd_resp_send_chunk(req, (const char *)buf, pad);
1269 }
1270 }
1271
1272 closedir(dir);
1273
1274 memset(buf, 0, 512);
1275 httpd_resp_send_chunk(req, (const char *)buf, 512);
1276 httpd_resp_send_chunk(req, (const char *)buf, 512);
1277
1278 httpd_resp_send_chunk(req, NULL, 0);
1279 net_mgr_notify_eth_activity();
1280 return ESP_OK;

1365 }
1366
1367 // Move to next entry (header + padded data)

1457 {

1462 esp_err_t sret = httpd_ws_send_frame_async(http_server, client_fds[i], &ws_pkt);
1463 LOGD(TAG, "ws_async_send: sent to client_fd[%d]=%d, sret=%d", i, client_fds[i], sret);

1465 }
