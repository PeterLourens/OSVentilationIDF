24 (void)pvParameters;

27 const TickType_t delay_ticks = pdMS_TO_TICKS(SYS_STATS_POLL_MS);

29 for (;;)

34 // Optionally publish or trigger any dependent work here (e.g., update web UI)
35 // ws_async_notify_stats_changed(); // if you have such a helper

37 vTaskDelay(delay_ticks);

40 // never reached
41 vTaskDelete(NULL);
42}

44// Call once at startup to create the task. Safe to call multiple times (idempotent).
45void start_task_sys_stats(void)
46{
47 if (task_sys_stats != NULL)
48 {
49 LOGI(TAG, "sys_stats_start: already running (task=%p)", (void *)task_sys_stats);
50 }
51
52 task_sys_stats = xTaskCreateStatic(task_sys_stats_code, "sys_stats", sys_stats_stack_size, NULL, 8, sys_stats_stack, &sys_stats_tcb);
53
54 if (task_sys_stats == NULL)
55 {
56 LOGE(TAG, "sys_stats_start: failed to create task");
57 task_sys_stats = NULL;
58 }
59
60 LOGI(TAG, "sys_stats_start: started (task=%p)", (void *)task_sys_stats);
61}
