93esp_err_t init_nvs(void)
94{
95 esp_err_t err = ESP_OK;
96
97 err = nvs_flash_init();
98 if (err == ESP_ERR_NVS_NO_FREE_PAGES || err == ESP_ERR_NVS_NEW_VERSION_FOUND)
99 {
100 ESP_LOGW(TAG, "NVS needs erase, retrying...");
101 ESP_ERROR_CHECK(nvs_flash_erase());
102 err = nvs_flash_init();
103 }
104
105 if (err != ESP_OK)
106 {
107 ESP_LOGE(TAG, "Failed to initialize NVS (%s)", esp_err_to_name(err));
108 return err;
109 }
110
111 ESP_LOGI(TAG, "NVS initialized");
112 return ESP_OK;

123esp_err_t init_littlefs(void)
124{
125 esp_err_t err = ESP_OK;
126
127 const esp_vfs_littlefs_conf_t conf = {
128 .base_path = "/data",
129 .partition_label = "storage",
130 .format_if_mount_failed = false,
131 .dont_mount = false,
132 };
133
134 err = esp_vfs_littlefs_register(&conf);
135 if (err != ESP_OK)
136 {
137 ESP_LOGE(TAG, "Failed to mount LittleFS (%s)", esp_err_to_name(err));
138 return err;
139 }
140
141 size_t total = 0;
142 size_t used = 0;
143
144 err = esp_littlefs_info(conf.partition_label, &total, &used);
145 if (err != ESP_OK)
146 {
147 ESP_LOGE(TAG, "Failed to get LittleFS partition info (%s)", esp_err_to_name(err));
148 esp_vfs_littlefs_unregister(conf.partition_label);
149 return err;
150 }
151
152 ESP_LOGI(TAG, "LittleFS mounted. Total: %d bytes, Used: %d bytes", total, used);
153
154 DIR *dir = opendir("/data");
155 if (!dir)
156 {
157 ESP_LOGE("LITTLEFS", "Failed to open /data");
158 return ESP_FAIL;
159 }
160
161 struct dirent *entry;
162 char full_path[270];
163 struct stat st;
164
165 while ((entry = readdir(dir)) != NULL)
166 {
167 snprintf(full_path, sizeof(full_path), "/data/%s", entry->d_name);
168
169 if (stat(full_path, &st) == 0)
170 {
171 ESP_LOGI("LITTLEFS", "Found: %s (%ld bytes)", entry->d_name, st.st_size);
172 }
173 else
174 {
175 ESP_LOGW("LITTLEFS", "Found: %s (size unknown)", entry->d_name);
176 }
177 }
178
179 closedir(dir);
180 return ESP_OK;
181}

191esp_err_t read_file(const char *path, char *buffer, size_t bufferSize)
192{
193 if (!path || !buffer || bufferSize == 0)
194 {
195 ESP_LOGE(TAG, "Invalid arguments");
196 return ESP_ERR_INVALID_ARG;
197 }
198
199 FILE *file = fopen(path, "rb"); // Use binary mode for safety
200 if (!file)
201 {
202 ESP_LOGE(TAG, "Failed to open file: %s", path);
203 return ESP_FAIL;
204 }
205
206 size_t len = fread(buffer, 1, bufferSize - 1, file);
207 buffer[len] = '\0';
208 fclose(file);
209
210 if (len == 0)
211 {
212 ESP_LOGW(TAG, "Read 0 bytes from file: %s", path);
213 return ESP_ERR_INVALID_SIZE;
214 }
215
216 ESP_LOGD(TAG, "Read %zu bytes from %s", len, path);
217 return ESP_OK;
218}

406void init_timezone(void)
407{
408 char ntp_server[64];
409 char timezone[64];
410
411 // Read config under mutex
412 if (xSemaphoreTake(time_config_mutex, pdMS_TO_TICKS(20)) != pdTRUE)
413 {
414 LOGE(TAG, "Failed to acquire time_config mutex");
415 return;
416 }
417
418 strncpy(ntp_server, time_config.ntp_server, sizeof(ntp_server));
419 ntp_server[sizeof(ntp_server) - 1] = '\0';
420 strncpy(timezone, time_config.timezone, sizeof(timezone));
421 timezone[sizeof(timezone) - 1] = '\0';
422 xSemaphoreGive(time_config_mutex);
423
424 setenv("TZ", "CET-1CEST,M3.5.0,M10.5.0/3", 1);
425 tzset();
426}

434void formatted_uptime(char *buf, size_t bufsize)
435{
436 uint64_t uptime = esp_timer_get_time() / 1000000; // uptime in seconds
437
438 int uptime_day = uptime / (60 * 60 * 24);
439 int uptime_hour = (uptime / (60 * 60)) % 24;
440 int uptime_minute = (uptime / 60) % 60;
441 int uptime_second = uptime % 60;
442
443 snprintf(buf, bufsize, "%dd %dh %02dm %02ds", uptime_day, uptime_hour, uptime_minute, uptime_second);
444}

453{

490void formatted_day(char *buf, size_t bufsize)
491{
492 int dayofweek = 0;
493 static const char *day_names[] = {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};
494 struct tm temp_time;
495 time_t now = 0;
496
497 time(&now);
498 localtime_r(&now, &temp_time);
499
500 // Get day of week from local time struct
501 dayofweek = temp_time.tm_wday;
502
503 // Clamp dayofweek to valid range just in case
504 if (dayofweek < 0 || dayofweek > 6)
505 {
506 dayofweek = 0;
507 }
508
509 snprintf(buf, bufsize, "%s", day_names[dayofweek]);
510}
