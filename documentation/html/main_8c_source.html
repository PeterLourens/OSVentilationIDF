93void app_main(void)
94{
95 esp_reset_reason_t reason = esp_reset_reason();
96 ESP_LOGI("boot", "Reset reason: %d", reason);
97 ESP_LOGI("boot", "Printing saved backtrace (if any):");
98 // esp_backtrace_print(ESP_LOG_INFO);
99
100 // dump task list and free heap
101 char tbuf[1024];
102 vTaskList(tbuf);
103 ESP_LOGI("boot", "Task list:\n%s", tbuf);
104 ESP_LOGI("boot", "Free heap: %u", esp_get_free_heap_size());
105
106 UBaseType_t high = uxTaskGetStackHighWaterMark(NULL);
107 ESP_LOGI("boot", "Main task stack high water mark: %u", (unsigned)high);
108
109 esp_err_t err = ESP_OK;
110
111 // Initialize hardware capabilities based on hardware version defined in hardware_caps.c
112 hw_init();
113
114 // Reduce cJSON internal heap usage by directing cJSON allocations into SPIRAM where available.
115 init_cjson_hooks();
116
117 // Initialize lifecycle mutex for Ethernet operations
118 eth_lifecycle_init_once();
119
120 // Set log levels for each TAG
121 set_log_levels();
122
123 // Install GPIO ISR service (required for W5500 interrupt pin, even if not used)
124 ESP_ERROR_CHECK(gpio_install_isr_service(ESP_INTR_FLAG_IRAM));
125
126 // Create mutexes
127 err = create_mutexes();
128 if (err != ESP_OK)
129 {
130 LOGE(TAG, "Failed to create mutexes");
131 return;
132 }
133
134 vTaskDelay(pdMS_TO_TICKS(1000));
135
136 // Initialize NVS
137 err = init_nvs();
138 if (err != ESP_OK)
139 {
140 LOGE(TAG, "Failed to initialize NVS");
141 return;
142 }
143
144 // Initialize LittleFS
145 err = init_littlefs();
146 if (err != ESP_OK)
147 {
148 LOGE(TAG, "Failed to initialize LittleFS");
149 return;
150 }
151
152 // init I2C bus0 and bus1
153 err = i2c_init_master0();
154 if (err != ESP_OK)
155 {
156 LOGE(TAG, "I2C master bus0 initialization failed: %s", esp_err_to_name(err));
157 return;
158 }
159 err = i2c_init_master1();
160 if (err != ESP_OK)
161 {
162 LOGE(TAG, "I2C master bus1 initialization failed: %s", esp_err_to_name(err));
163 return;
164 }
165
166 // Init LCD on bus1
167 err = i2c_lcd_init(i2c_bus1_handle, &lcd_bus1_handle);
168 if (err != ESP_OK)
169 {
170 LOGE(TAG, "Failed to initialize LCD on bus 1: %s", esp_err_to_name(err));
171 vTaskDelete(NULL);
172 return;
173 }
174 else
175 {
176 LOGI(TAG, "LCD initialized on bus 1");
177 }
178
179 // Load sensor configuration for bus 0
180 err = load_sensor_config("bus0_sensor_config");
181 if (err != ESP_OK)
182 {
183 LOGE(TAG, "Failed to load bus 0 sensor configuration");
184 return;
185 }
186
187 // Load sensor configuration for bus 1
188 err = load_sensor_config("bus1_sensor_config");
189 if (err != ESP_OK)
190 {
191 LOGE(TAG, "Failed to load bus 1 sensor configuration");
192 return;
193 }
194
195 // Load wifi configuration
196 err = load_wifi_config();
197 if (err != ESP_OK)
198 {
199 LOGE(TAG, "Failed to load wifi configuration");
200 return;
201 }
202
203 // Load ethernet configuration
204 err = load_ethernet_config();
205 if (err != ESP_OK)
206 {
207 LOGE(TAG, "Failed to load ethernet configuration");
208 return;
209 }
210
211 // Load time configuration
212 err = load_time_config();
213 if (err != ESP_OK)
214 {
215 LOGE(TAG, "Failed to load time configuration");
216 return;
217 }
218
219 // set timezone after reading time config
220 init_timezone();
221
222 // Load MQTT configuration
223 err = load_mqtt_config();
224 if (err != ESP_OK)
225 {
226 LOGE(TAG, "Failed to load MQTT configuration");
227 return;
228 }
229
230 // Load InfluxDB configuration
231 err = load_influxdb_config();
232 if (err != ESP_OK)
233 {
234 LOGE(TAG, "Failed to load InfluxDB configuration");
235 return;
236 }
237
238 // Load fan control configuration
239 err = load_fan_config();
240 if (err != ESP_OK)
241 {
242 LOGE(TAG, "Failed to load fan control configuration");
243 return;
244 }
245
246 // Load I2C control configuration
247 err = load_i2c_config();
248 if (err != ESP_OK)
249 {
250 LOGE(TAG, "Failed to load I2C configuration");
251 return;
252 }
253
254 // Load syslog control configuration
255 err = load_syslog_config();
256 if (err != ESP_OK)
257 {
258 LOGE(TAG, "Failed to load syslog configuration");
259 return;
260 }
261
262 // Start UART sender task early so we can log from the start (including hw_init)
263 if (strcmp(syslog_config.enable_uart, "On") == 0)
264 {
265 start_task_uart_sender();
266 }
267 else
268 {
269 LOGI(TAG, "UART syslog forwarding disabled by configuration");
270 }
271
272 // Load statemachine general configuration
273 err = load_statemachine_general_config();
274 if (err != ESP_OK)
275 {
276 LOGE(TAG, "Failed to load statemachine general configuration");
277 return;
278 }
279
280 // Load state configuration
281 err = load_state_config();
282 if (err != ESP_OK)
283 {
284 LOGE(TAG, "Failed to load states configuration");
285 return;
286 }
287
288 // Read valve positions from NVS into valvecontrol struct
289 read_valve_pos_in_valvecontrol();
290
291 // Iinit influxdb sender task and queue
292 if (!influxdb_init_sender())
293 {
294 LOGW("main", "InfluxDB sender init failed; Influx disabled for this boot");
295 }
296
297 // TCP/IP stack and default event loop (do once)
298 ESP_ERROR_CHECK(esp_netif_init());
299 ESP_ERROR_CHECK(esp_event_loop_create_default());
300
301 // Start tasks
302 LOGI(TAG, "Starting system tasks...");
303 syslog_queue_init(); // must be before network services to capture logs from startup
304 network_manager_start(); // starts network manager task which starts Wi‑Fi and Ethernet as needed
305 if (strcmp(syslog_config.enable_uart, "On") == 0)
306 {
307 start_task_uart_sender();
308 }
309 else
310 {
311 LOGI(TAG, "UART syslog forwarding disabled by configuration");
312 }
313 start_task_i2c_ds3231();
314 start_task_i2c_sensors();
315 start_task_i2c_lcd();
316 start_task_rgb_leds();
317 start_task_valvecontrol();
318 start_task_pushbutton();
319 start_task_sys_stats();
320 // ws_broadcaster_init();
321 // stats_worker_init();
322
323 // Start network services in a separate task so we can retry if no network at startup
324 start_network_services();
325
326 // Initialize sensors based on current configuration
327 reload_and_init_sensors();
328
329 esp_log_set_vprintf(uart_log_vprintf);
330 LOGI(TAG, "System startup, log output is now routed through uart_log_vprintf");
331
332 while (1)
333 {
334 char uptime_str[64] = {};
335
336 formatted_uptime(uptime_str, sizeof(uptime_str));
337 LOGI(TAG, "System uptime: %s", uptime_str);
338
339 // sys_stats_update_actuals();
340 update_hostname();
341 LOGI(TAG, "Updated system stats");
342
343 // Ensure connected and have IP before pinging (check esp_netif_get_ip_info)
344 if (!wifi_has_ip())
345 {
346 if (ping_gateway())
347 {
348 LOGI(TAG, "gateway OK");
349 }
350 else
351 {
352 LOGW(TAG, "gateway unreachable — consider reconnecting Wi‑Fi");
353 }
354 }
355
356 vTaskDelay(pdMS_TO_TICKS(600000));
357 }
358}
