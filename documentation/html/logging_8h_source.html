23#define LOG_FORMAT_AND_SEND(tag, level, prefix, fmt, ...) \
24 do \
25 { \
26 /* Always log locally */ \
27 ESP_LOG_LEVEL(level, tag, fmt, ##__VA_ARGS__); \
28 \
29 /* Respect ESP-IDF log level for syslog too */ \
30 if (level > esp_log_level_get(tag)) \
31 break; \
32 \
33 if (!syslog_enabled) \
34 break; \
35 \
36 /* First pass: compute required size */ \
37 int _len = snprintf(NULL, 0, fmt, ##__VA_ARGS__); \
38 if (_len < 0) \
39 break; /* formatting error */ \
40 \
41 /* Allocate exact buffer */ \
42 char *_msg = heap_caps_malloc(_len + 1, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT); \
43 if (!_msg) \
44 break; \
45 \
46 /* Second pass: write formatted message */ \
47 snprintf(_msg, _len + 1, fmt, ##__VA_ARGS__); \
48 \
49 /* Prefix for syslog */ \
50 int _plen = snprintf(NULL, 0, "[%s] %s", prefix, _msg); \
51 if (_plen < 0) \
52 { \
53 heap_caps_free(_msg); \
54 break; \
55 } \
56 \
57 char *_sys = heap_caps_malloc(_plen + 1, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT); \
58 if (!_sys) \
59 { \
60 heap_caps_free(_msg); \
61 break; \
62 } \
63 \
64 snprintf(_sys, _plen + 1, "[%s] %s", prefix, _msg); \
65 \
66 syslog_send_line(_sys); \
67 \
68 heap_caps_free(_msg); \
69 heap_caps_free(_sys); \
70 } while (0)
