54// Initialize pool free list; must be called before using pool
55static void uart_pool_init(void)
56{
57 if (uart_pool_lock == NULL)
58 {
59 uart_pool_lock = xSemaphoreCreateMutex();
60 if (uart_pool_lock == NULL)
61 {
62 ESP_LOGE(TAG, "Failed to create uart_pool_lock");
63 return;
64 }
65 }
66
67 if (uart_pool_buf == NULL)
68 {
69 size_t total_size = (size_t)UART_POOL_COUNT * UART_POOL_BUF_SIZE;
70 uart_pool_buf = heap_caps_malloc(total_size, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
71 if (uart_pool_buf == NULL)
72 {
73 ESP_LOGW(TAG, "uart_pool_init: SPIRAM allocation failed, falling back to internal RAM");
74 uart_pool_buf = malloc(total_size);
75 if (uart_pool_buf == NULL)
76 {
77 ESP_LOGE(TAG, "uart_pool_init: failed to allocate uart_pool_buf");
78 return;
79 }
80 }
81 }
82
83 xSemaphoreTake(uart_pool_lock, portMAX_DELAY);
84 uart_pool_free_top = 0;
85 for (int i = 0; i < UART_POOL_COUNT; ++i)
86 {
87 uart_pool_free_idx[uart_pool_free_top++] = i;
88 }
89 xSemaphoreGive(uart_pool_lock);
90}

92// Non-blocking allocate: returns pointer to buffer or NULL if none available
93static char *uart_pool_alloc(void)
94{
95 char *ret = NULL;
96 if (uart_pool_lock == NULL || uart_pool_buf == NULL)
97 return NULL;
98
99 if (xSemaphoreTake(uart_pool_lock, 0) == pdTRUE)
100 {
101 if (uart_pool_free_top > 0)
102 {
103 int idx = uart_pool_free_idx[--uart_pool_free_top];
104 ret = (char *)(uart_pool_buf + ((size_t)idx * UART_POOL_BUF_SIZE));
105 }
106 xSemaphoreGive(uart_pool_lock);
107 }
108 return ret;
109}

111// Free buffer previously allocated from pool
112static void uart_pool_free(char *buf)
113{
114 if (buf == NULL || uart_pool_lock == NULL)
115 return;
116
117 // compute index
118 ptrdiff_t offset = buf - (char *)uart_pool_buf;
119 if (offset < 0)
120 return;
121 int idx = offset / UART_POOL_BUF_SIZE;
122 if (idx < 0 || idx >= UART_POOL_COUNT)
123 return;
124
125 if (xSemaphoreTake(uart_pool_lock, portMAX_DELAY) == pdTRUE)
126 {
127 uart_pool_free_idx[uart_pool_free_top++] = idx;
128 xSemaphoreGive(uart_pool_lock);
129 }
130}

132int uart_log_vprintf(const char *fmt, va_list args)
133{
134 if (fmt == NULL)
135 return 0;
136
137 char local_buf[UART_POOL_BUF_SIZE];
138 int n = vsnprintf(local_buf, UART_POOL_BUF_SIZE - 3, fmt, args);
139 if (n < 0)
140 return 0;
141
142 if (n > UART_POOL_BUF_SIZE - 3)
143 {
144 n = UART_POOL_BUF_SIZE - 3;
145 local_buf[n] = '\0';
146 }
147 else
148 {
149 local_buf[n] = '\0';
150 }
151
152 // Keep the original log output on the JTAG/serial monitor / console
153 if (n > 0)
154 {
155 fwrite(local_buf, 1, n, stdout);
156 fflush(stdout);
157 }
158
159 if (uart_tx_queue == NULL)
160 {
161 return n;
162 }
163
164 // Allocate buffer from pool (non-blocking)
165 char *buf = uart_pool_alloc();
166 if (!buf)
167 return n;
168
169 memcpy(buf, local_buf, n + 1);
170
171 // Parse ESP-IDF log format and extract tag:message without timestamp
172 char *parsed_msg = buf;
173 int parsed_len = n;
174
175 // Skip ANSI escape codes at the beginning
176 char *start = buf;
177 while (start < buf + n && *start == '\033')
178 {
179 char *m = strchr(start, 'm');
180 if (m)
181 {
182 start = m + 1;
183 }
184 else
185 {
186 break;
187 }
188 }
189
190 // ESP-IDF format: "L (timestamp) tag: message"
191 if (start < buf + n && *start >= 'A' && *start <= 'Z' && start[1] == ' ' && start[2] == '(')
192 {
193 // Find the closing ) after timestamp
194 char *close_paren = strchr(start + 2, ')');
195 if (close_paren && close_paren[1] == ' ')
196 {
197 // Skip "L (timestamp) " and keep "tag: message"
198 parsed_msg = close_paren + 2;
199 parsed_len = n - (parsed_msg - buf);
200 }
201 }
202 else
203 {
204 // Not a standard ESP-IDF log, use the message after skipping ANSI codes
205 parsed_msg = start;
206 parsed_len = n - (start - buf);
207 }
208
209 // Skip if message is empty or just whitespace/newlines
210 if (parsed_len <= 1 || (parsed_len == 2 && parsed_msg[0] == '\r' && parsed_msg[1] == '\n'))
211 {
212 uart_pool_free(buf);
213 return n; // Return original length for ESP-IDF
214 }
215
216 // Trim trailing whitespace/newlines
217 while (parsed_len > 0 && (parsed_msg[parsed_len - 1] == '\n' || parsed_msg[parsed_len - 1] == '\r' || parsed_msg[parsed_len - 1] == ' '))
218 {
219 parsed_len--;
220 }
221
222 if (parsed_len <= 0)
223 {
224 uart_pool_free(buf);
225 return n;
226 }
227
228 // Copy parsed message back to buffer
229 memmove(buf, parsed_msg, parsed_len);
230 buf[parsed_len] = '\0';
231
232 // Add CRLF for UART
233 if (parsed_len + 2 < UART_POOL_BUF_SIZE)
234 {
235 buf[parsed_len++] = '\r';
236 buf[parsed_len++] = '\n';
237 buf[parsed_len] = '\0';
238 }
239
240 // Enqueue pointer; non-blocking
241 BaseType_t ok = xQueueSend(uart_tx_queue, &buf, 0);
242 if (ok != pdTRUE)
243 {
244 uart_pool_free(buf);
245 return 0;
246 }
247
248 return n;
249}

252static esp_err_t uart_sender_hw_init(void)
253{
254 ESP_LOGI(TAG, "hw_init: start");
255
256 if (hw.uart0_tx_gpio < 0 || hw.uart0_rx_gpio < 0)
257 {
258 ESP_LOGE(TAG, "hw_init: invalid pins tx=%d rx=%d", hw.uart0_tx_gpio, hw.uart0_rx_gpio);
259 return ESP_ERR_INVALID_ARG;
260 }
261
262 uart_config_t uart_config = {
263 .baud_rate = SENDER_UART_BAUD,
264 .data_bits = UART_DATA_8_BITS,
265 .parity = UART_PARITY_DISABLE,
266 .stop_bits = UART_STOP_BITS_1,
267 .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
268 .source_clk = UART_SCLK_APB,
269 };
270
271 ESP_LOGI(TAG, "uart_sender_hw_init: configuring UART%d (TX=%d RX=%d) @%d", SENDER_UART_NUM, hw.uart0_tx_gpio, hw.uart0_rx_gpio, SENDER_UART_BAUD);
272
273 // Configure UART parameters first (safe to call before driver install)
274 esp_err_t err = uart_param_config(SENDER_UART_NUM, &uart_config);
275 if (err != ESP_OK)
276 {
277 ESP_LOGE(TAG, "uart_param_config failed: %s", esp_err_to_name(err));
278 return err;
279 }
280
281 // Set pins (use GPIO numbers, not physical pad indices)
282 err = uart_set_pin(SENDER_UART_NUM, hw.uart0_tx_gpio, hw.uart0_rx_gpio, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE);
283 if (err != ESP_OK)
284 {
285 ESP_LOGE(TAG, "uart_set_pin failed: %s", esp_err_to_name(err));
286 return err;
287 }
288
289 // Small delay to let pin matrix settle before driver install
290 vTaskDelay(pdMS_TO_TICKS(10));
291
292 // Install driver. Try default flags first; if it fails, try IRAM flag (if available).
293 ESP_LOGI(TAG, "uart_sender_hw_init: installing driver rx=%d tx=%d", SENDER_UART_RX_BUF_SIZE, SENDER_TX_BUF_SIZE);
294
295 err = uart_driver_install(SENDER_UART_NUM, SENDER_UART_RX_BUF_SIZE, SENDER_TX_BUF_SIZE, 0, NULL, 0);
296
297 if (err != ESP_OK)
298 {
299 ESP_LOGW(TAG, "uart_driver_install(default) failed: %s", esp_err_to_name(err));
300#ifdef ESP_INTR_FLAG_IRAM
301 ESP_LOGI(TAG, "uart_sender_hw_init: retrying uart_driver_install with ESP_INTR_FLAG_IRAM");
302 err = uart_driver_install(SENDER_UART_NUM, SENDER_UART_RX_BUF_SIZE, SENDER_TX_BUF_SIZE, 0, NULL, ESP_INTR_FLAG_IRAM);
303 if (err != ESP_OK)
304 {
305 ESP_LOGE(TAG, "uart_driver_install(IRAM) failed: %s", esp_err_to_name(err));
306 return err;
307 }
308#else
309 return err;
310#endif
311 }
312
313 // Mark driver installed for runtime checks
314 uart_driver_installed_flag = true;
315
316 // Small delay to allow driver/interrupts to settle
317 vTaskDelay(pdMS_TO_TICKS(20));
318
319 ESP_LOGI(TAG, "UART initialized UART%d TX=%d RX=%d @%d", SENDER_UART_NUM, hw.uart0_tx_gpio, hw.uart0_rx_gpio, SENDER_UART_BAUD);
320
321 return ESP_OK;
322}

324// -------------------- Public send API (pool-backed) -------------------- //
325void uart_sender_sendf(const char *fmt, ...)
326{
327 if (fmt == NULL)
328 return;
329
330 // If queue not created yet, drop message
331 if (uart_tx_queue == NULL)
332 {
333 uart_sender_dropped_msgs++;
334 return;
335 }
336
337 // Allocate buffer from pool (non-blocking)
338 char *buf = uart_pool_alloc();
339 if (!buf)
340 {
341 uart_sender_pool_exhaust++;
342 uart_sender_dropped_msgs++;
343 return;
344 }
345
346 va_list ap;
347 va_start(ap, fmt);
348 int n = vsnprintf(buf, UART_POOL_BUF_SIZE - 3, fmt, ap);
349 va_end(ap);
350
351 if (n < 0)
352 {
353 // return buffer to pool
354 uart_pool_free(buf);
355 return;
356 }
357
358 if (n > UART_POOL_BUF_SIZE - 3)
359 {
360 n = UART_POOL_BUF_SIZE - 3;
361 buf[n] = '\0';
362 }
363
364 // Append CRLF
365 buf[n++] = '\r';
366 buf[n++] = '\n';
367 buf[n] = '\0';
368
369 // Enqueue pointer; non-blocking
370 BaseType_t ok = xQueueSend(uart_tx_queue, &buf, 0);
371 if (ok != pdTRUE)
372 {
373 // queue full: return buffer to pool and count drop
374 uart_pool_free(buf);
375 uart_sender_dropped_msgs++;
376 }
377}

379// -------------------- TX worker (diagnostic, defensive) -------------------- //
380static void uart_tx_task(void *pvParameters)
381{
382 (void)pvParameters;
383
384 if (uart_tx_queue == NULL)
385 {
386 ESP_LOGE(TAG, "uart_tx_task: uart_tx_queue is NULL; exiting");
387 vTaskDelete(NULL);
388 return;
389 }
390
391 char *line = NULL;
392 TickType_t last_report = xTaskGetTickCount();
393
394 for (;;)
395 {
396 // Periodic diagnostics every 5s
397 if ((xTaskGetTickCount() - last_report) > pdMS_TO_TICKS(5000))
398 {
399 UBaseType_t high = uxTaskGetStackHighWaterMark(NULL);
400 ESP_LOGD(TAG, "TX task HWM=%u free_heap=%u pool_free=%d dropped=%u pool_exhaust=%u",
401 (unsigned)high,
402 (unsigned)esp_get_free_heap_size(),
403 uart_pool_free_top,
404 (unsigned)uart_sender_dropped_msgs,
405 (unsigned)uart_sender_pool_exhaust);
406 last_report = xTaskGetTickCount();
407 }
408
409 if (xQueueReceive(uart_tx_queue, &line, portMAX_DELAY) == pdTRUE)
410 {
411 if (!line)
412 {
413 ESP_LOGW(TAG, "uart_tx_task: received NULL pointer");
414 continue;
415 }
416
417 size_t len = strlen(line);
418 int written = uart_write_bytes(SENDER_UART_NUM, line, len);
419 if (written < 0)
420 {
421 ESP_LOGW(TAG, "uart_tx_task: uart_write_bytes error %d", written);
422 }
423 else if ((size_t)written != len)
424 {
425 ESP_LOGW(TAG, "uart_tx_task: partial write %d/%u", written, (unsigned)len);
426 }
427
428 // Wait for TX to finish at driver level (timeout 2s)
429 esp_err_t wait_err = uart_wait_tx_done(SENDER_UART_NUM, pdMS_TO_TICKS(2000));
430 if (wait_err != ESP_OK)
431 {
432 ESP_LOGW(TAG, "uart_tx_task: uart_wait_tx_done %s", esp_err_to_name(wait_err));
433 }
434
435 // Return buffer to pool
436 uart_pool_free(line);
437 line = NULL;
438
439 // Yield briefly so interrupts and watchdog can run
440 vTaskDelay(pdMS_TO_TICKS(1));
441 }
442 }
443
444 vTaskDelete(NULL);
445}

462// -------------------- Start function --------------------
463void start_task_uart_sender(void)
464{
465 // Initialize pool first
466 uart_pool_init();
467
468 // Initialize hardware first
469 if (uart_sender_hw_init() != ESP_OK)
470 {
471 ESP_LOGE(TAG, "UART hardware init failed; not starting sender tasks");
472 return;
473 }
474
475 // Create queue for pointers to pool buffers
476 if (uart_tx_queue == NULL)
477 {
478 uart_tx_queue = xQueueCreate(UART_SENDER_QUEUE_LEN, sizeof(char *));
479 if (uart_tx_queue == NULL)
480 {
481 ESP_LOGE(TAG, "Failed to create uart_tx_queue");
482 uart_driver_delete(SENDER_UART_NUM);
483 return;
484 }
485 }
486
487 // Create TX worker task (pinned to core 0, low priority)
488 if (task_uart_tx_sender == NULL)
489 {
490 BaseType_t r = xTaskCreatePinnedToCore(uart_tx_task, "uart_tx_task", 8192, NULL, tskIDLE_PRIORITY + 1, &task_uart_tx_sender, 1);
491 if (r != pdPASS)
492 {
493 ESP_LOGE(TAG, "Failed to create uart_tx_task");
494 task_uart_tx_sender = NULL;
495 if (uart_tx_queue)
496 {
497 vQueueDelete(uart_tx_queue);
498 uart_tx_queue = NULL;
499 }
500 uart_driver_delete(SENDER_UART_NUM);
501 return;
502 }
503 }
504
505 // Create the user-visible sender task as a static task (one-time)
506 if (task_uart_sender == NULL)
507 {
508 task_uart_sender = xTaskCreateStatic(task_uart_sender_code, "uart_sender", UART_SENDER_STACK_WORDS, NULL, tskIDLE_PRIORITY + 1, uart_sender_stack, &uart_sender_tcb);
509
510 if (task_uart_sender == NULL)
511 {
512 ESP_LOGE(TAG, "UART sender task creation failed");
513 if (task_uart_tx_sender)
514 {
515 vTaskDelete(task_uart_tx_sender);
516 task_uart_tx_sender = NULL;
517 }
518 if (uart_tx_queue)
519 {
520 vQueueDelete(uart_tx_queue);
521 uart_tx_queue = NULL;
522 }
523 uart_driver_delete(SENDER_UART_NUM);
524 return;
525 }
526 }
527
528 ESP_LOGI(TAG, "UART sender started (tx_handle=%p sender_handle=%p pool_count=%d buf_size=%d)", (void *)task_uart_tx_sender, (void *)task_uart_sender, UART_POOL_COUNT, UART_POOL_BUF_SIZE);
529}
