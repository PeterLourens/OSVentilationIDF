46// Task code: uses ISR queue when available, falls back to polling for robustness
47void task_pushbutton_code(void *pvParameters)
48{
49 (void)pvParameters;
50 pushbutton_init();
51
52 gpio_num_t pin = get_pushbutton_gpio_local();
53 QueueHandle_t q = pushbutton_get_queue();
54
55 TickType_t last_press_tick = 0;
56 bool pressed_state = (gpio_get_level(pin) == 0); // active low
57
58 for (;;)
59 {
60 // Prefer event-driven path if queue exists
61 if (q)
62 {
63 uint32_t evt;
64 if (xQueueReceive(q, &evt, pdMS_TO_TICKS(POLL_MS)) == pdTRUE)
65 {
66 // Debounce: wait and re-read
67 vTaskDelay(pdMS_TO_TICKS(DEBOUNCE_MS));
68 int level = gpio_get_level(pin);
69 bool pressed = (level == 0);
70 if (pressed && !pressed_state)
71 {
72 // pressed edge
73 last_press_tick = xTaskGetTickCount();
74 pressed_state = true;
75 }
76 else if (!pressed && pressed_state)
77 {
78 // release edge
79 if (last_press_tick != 0)
80 {
81 TickType_t now = xTaskGetTickCount();
82 uint32_t ms = (uint32_t)((now - last_press_tick) * portTICK_PERIOD_MS);
83 if (ms < LONG_PRESS_MS)
84 {
85 toggle_backlight();
86 }
87 else
88 {
89 LOGI(TAG, "Button long press %lums (no action)", (unsigned long)ms);
90 }
91 last_press_tick = 0;
92 }
93 pressed_state = false;
94 }
95 }
96 else
97 {
98 // timeout: allow low-power polling; continue loop
99 }
100 }
101 else
102 {
103 // Fallback polling if queue not available
104 vTaskDelay(pdMS_TO_TICKS(POLL_MS));
105 int level = gpio_get_level(pin);
106 bool pressed = (level == 0);
107 if (pressed != pressed_state)
108 {
109 // potential edge: debounce
110 vTaskDelay(pdMS_TO_TICKS(DEBOUNCE_MS));
111 level = gpio_get_level(pin);
112 bool stable = (level == 0);
113 if (stable == pressed)
114 {
115 if (pressed)
116 {
117 last_press_tick = xTaskGetTickCount();
118 }
119 else
120 {
121 if (last_press_tick != 0)
122 {
123 TickType_t now = xTaskGetTickCount();
124 uint32_t ms = (uint32_t)((now - last_press_tick) * portTICK_PERIOD_MS);
125 if (ms < LONG_PRESS_MS)
126 toggle_backlight();
127 else
128 LOGI(TAG, "Button long press %lums (no action)", (unsigned long)ms);
129 last_press_tick = 0;
130 }
131 }
132 pressed_state = pressed;
133 }
134 }
135 }
136 }
137}

139// Start/stop helpers
140void start_task_pushbutton(void)
141{
142 if (task_pushbutton != NULL)
143 {
144 LOGW(TAG, "pushbutton task already running");
145 return;
146 }
147
148 task_pushbutton = xTaskCreateStatic(task_pushbutton_code, "pushbtn", pushbutton_stack_size, NULL, 9, pushbutton_stack, &pushbutton_tcb);
149 if (!task_pushbutton)
150 LOGE(TAG, "Failed to create pushbutton task");
151 else
152 LOGI(TAG, "Pushbutton task started");
153}
154
155void stop_task_pushbutton(void)
156{
157 if (task_pushbutton == NULL)
158 {
159 LOGI(TAG, "Pushbutton task not running");
160 return;
161 }
162
163 pushbutton_deinit();
164 vTaskDelete(task_pushbutton);
165 task_pushbutton = NULL;
166 LOGI(TAG, "Pushbutton task stopped");
167}
