53ColorName_t state_to_color(int state)
54{
55 switch (state)
56 {
57 case 0:
58 return COLOR_RED; // Error
59 case 1:
60 return COLOR_GREEN; // Init
61 case 2:
62 return COLOR_YELLOW; // Day
63 case 3:
64 return COLOR_BLUE; // Night
65 case 4:
66 return COLOR_DEEP_SKY_BLUE; // High CO2 day
67 case 5:
68 return COLOR_SEA_GREEN; // High CO2 night
69 case 6:
70 return COLOR_DEEP_SKY_BLUE; // High RH day
71 case 7:
72 return COLOR_CHOCOLATE; // High RH night
73 case 8:
74 return COLOR_GOLD; // Cooking
75 case 9:
76 return COLOR_DEEP_ORANGE; // Cycling day
77 case 10:
78 return COLOR_HOT_PINK; // Cycling night
79 case 11:

81 case 12:

83 case 13:

85 case 14:

139void set_led_brightness(int led, ColorName_t color, uint8_t brightness)

161// BLINK TOGGLE STATE
162void toggle_led(int led, ColorName_t color, bool *state)
163{
164 *state = !*state;
165
166 if (*state)
167 set_led_brightness(led, color, led_brightness);
168 else

172// MAIN LED UPDATE (called by LED task)
173void leds_update(const Statemachine_actuals_t *sm)
174{
175 // Copy snapshot
176 sm_snap = *sm;
177
178 // LED0: STATE
179 if (sm_snap.state == 0)
180 {
181 toggle_led(0, COLOR_RED, &blink0);
182 }
183 else if (sm_snap.valve_lock)
184 {
185 toggle_led(0, state_to_color(sm_snap.state), &blink0);
186 }
187 else
188 {
189 blink0 = false;
190 set_led(0, state_to_color(sm_snap.state));
191 }
192
193 // LED1: Network status at startup, fanspeed with periodic blink when running
194 static bool startup_mode = true;
195 static bool blink_override = false;
196
197 static int startup_counter = 0;
198
199 static uint32_t blink_start_ms = 0;
200 static uint32_t last_blink_ms = 0;
201
202 // Detect end of startup: after 5 seconds or both network interfaces checked
203 if (startup_mode)
204 {
205 startup_counter++;
206 if (startup_counter > 25)
207 { // ~5s if leds_update called every 200ms
208 startup_mode = false;
209 }
210 }
211
212 uint32_t now = (uint32_t)(esp_timer_get_time() / 1000); // ms
213
214 const bool use_wifi_blink = (hw_version == HW_VERSION_2 && sm_snap.wifi_connected);
215 const bool skip_network_blink = (hw_version == HW_VERSION_3);
216 ColorName_t net_color = COLOR_RED;
217
218 if (use_wifi_blink)
219 {
220 LOGD(TAG, "Blink blue (WIFI: %s)", sm_snap.wifi_connected ? "connected" : "disconnected");
221 net_color = COLOR_BLUE;
222 }
223 else
224 {
225 net_color = COLOR_RED;
226 }
227
228 if (skip_network_blink)
229 {
230 set_led(1, fan_to_color(sm_snap.fanspeed));
231 }
232 else if (startup_mode)
233 {
234 // Blink network status on LED1
235 static bool blink_on = false;
236 static uint32_t last_toggle = 0;
237 if (now - last_toggle > 300)
238 { // 300ms period
239 blink_on = !blink_on;
240 last_toggle = now;
241 }
242 if (blink_on)
243 set_led(1, net_color);
244 else
245 set_off(1);
246 }
247 else
248 {
249 // Normal: fanspeed color, but blink network status every 30s
250 if (!blink_override && (now - last_blink_ms > 30000))
251 {
252 blink_override = true;
253 blink_start_ms = now;
254 last_blink_ms = now;
255 }
256 if (blink_override)
257 {
258 set_led(1, net_color);
259 if (now - blink_start_ms > 120)
260 { // 120ms blink
261 blink_override = false;
262 }
263 }
264 else
265 {
266 set_led(1, fan_to_color(sm_snap.fanspeed));
267 }
268 }
269
270 led_strip_refresh(rgb_leds);
271}
