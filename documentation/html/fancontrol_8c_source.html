53static int do_http_get(const char *url, char *out_buf, size_t out_buf_len)
54{
55 esp_http_client_config_t config = {
56 .url = url,
57 .method = HTTP_METHOD_GET,
58 };
59
60 esp_http_client_handle_t client = esp_http_client_init(&config);
61 if (client == NULL)
62 {
63 LOGE(TAG, "Failed to init HTTP client for %s", url);
64 return -1;
65 }
66
67 esp_err_t err = esp_http_client_perform(client);
68 if (err != ESP_OK)
69 {
70 LOGE(TAG, "HTTP GET request failed for %s: %s", url, esp_err_to_name(err));
71 esp_http_client_cleanup(client);
72 return -1;
73 }
74
75 int status = esp_http_client_get_status_code(client);
76
77 if (out_buf && out_buf_len > 0)
78 {
79 int r = esp_http_client_read(client, out_buf, out_buf_len - 1);
80 if (r > 0)
81 {
82 out_buf[r] = '\0';
83 }
84 else
85 {
86 out_buf[0] = '\0';
87 }
88 }
89
90 esp_http_client_cleanup(client);
91 return status;
92}

113
114 // Use integer fanspeed passed in (0=error,1=low,2=medium,3=high)
115 int cur_speed = fanspeed;

117 // Read fan control config under fan_config_mutex
118 if (fan_config_mutex && xSemaphoreTake(fan_config_mutex, (TickType_t)50) == pdTRUE)

120 strncpy(fan_control_mode, fan_config.fan_control_mode, sizeof(fan_control_mode) - 1);
121 fan_control_mode[sizeof(fan_control_mode) - 1] = '\0';
122
123 strncpy(fan_control_mqtt_topic, fan_config.fan_control_mqtt_topic, sizeof(fan_control_mqtt_topic) - 1);
124 fan_control_mqtt_topic[sizeof(fan_control_mqtt_topic) - 1] = '\0';
125
126 strncpy(fan_control_url_high_speed, fan_config.fan_control_url_hs, sizeof(fan_control_url_high_speed) - 1);
127 fan_control_url_high_speed[sizeof(fan_control_url_high_speed) - 1] = '\0';
128
129 strncpy(fan_control_url_medium_speed, fan_config.fan_control_url_ms, sizeof(fan_control_url_medium_speed) - 1);
130 fan_control_url_medium_speed[sizeof(fan_control_url_medium_speed) - 1] = '\0';
131
132 strncpy(fan_control_url_low_speed, fan_config.fan_control_url_ls, sizeof(fan_control_url_low_speed) - 1);
133 fan_control_url_low_speed[sizeof(fan_control_url_low_speed) - 1] = '\0';
134
135 xSemaphoreGive(fan_config_mutex);
136 }
137
138 LOGD(TAG, "Fanspeed %s", fanspeed_to_str(cur_speed));
139
140 if (strcmp(fan_control_mode, "MQTT publish") == 0)
141 {
142 LOGD(TAG, "Using MQTT to set fan speed");
143 // Not yet implemented: subscribe/publish to MQTT topic
144 }
145 else if (strcmp(fan_control_mode, "HTTP API") == 0)
146 {
147 LOGD(TAG, "Using HTTP API to set fan speed");
148
149 if (cur_speed == FANSPEED_LOW)
150 {
151 LOGD(TAG, "Low speed selected");
152 httpResponseCode = do_http_get(fan_control_url_low_speed, http_payload, sizeof(http_payload));
153 if (httpResponseCode > 0)
154 {
155 LOGD(TAG, "HTTP Response code: %d, HTTP payload: %s", httpResponseCode, http_payload);
156 }

158 {
159 LOGE(TAG, "HTTP request failed for Low speed (code %d)", httpResponseCode);
160 }
161 }

173 }
174 }

177 LOGD(TAG, "High speed selected");

179 if (httpResponseCode > 0)
