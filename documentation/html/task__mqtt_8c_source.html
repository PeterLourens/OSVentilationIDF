30void task_mqtt_code(void *pvParameters)
31{
32 (void)pvParameters;
33
34 // Give network_manager time to bring up interfaces and start MQTT
35 vTaskDelay(pdMS_TO_TICKS(3000));
36
37 LOGI(TAG, "MQTT task starting...");
38
39 while (true)
40 {
41 LOGI(TAG, "MQTT: publish cycle");
42
43 if (!mqtt_is_connected())
44 {
45 LOGW(TAG, "MQTT: not connected, skipping publish");
46 vTaskDelay(pdMS_TO_TICKS(5000));
47 continue;
48 }
49
50 // Handle deferred HA discovery here (MQTT task context)
51 if (mqtt_discovery_requested() && sensors_ready())
52 {
53 mqtt_clear_discovery_request();
54 mqtt_publish_homeassistant_discovery();
55 }
56
57 mqtt_publish_sensor_values();
58 mqtt_publish_valve_positions();
59 mqtt_publish_state_and_fanspeed();
60
61 vTaskDelay(pdMS_TO_TICKS(20000));
62 }
