34 const size_t snapshot_bytes = sizeof(float) * snap_count;
35
36 // Record item size globally so other tasks can sanity-check before peeking

39 ESP_LOGD(TAG, "task_i2c_sensors: snapshot_bytes=%zu, free_heap=%u", snapshot_bytes, esp_get_free_heap_size());

44
45 // Create static queues (1 slot each) to avoid large heap allocations
46 sensor_queue = xQueueCreateStatic(1, snapshot_bytes, sensor_queue_storage, &sensor_queue_struct);
47
48 if (sensor_queue == NULL)

50 ESP_LOGE(TAG, "xQueueCreateStatic(sensor_queue) failed (item size=%zu) free_heap=%u", snapshot_bytes, esp_get_free_heap_size());

52 vTaskDelete(NULL);

64 sensor_queue = NULL;

69
70 // Signal readiness for consumers
71 sensors_queues_ready = true;
72 ESP_LOGI(TAG, "sensor queues ready");
73
74 // Initialize both TCA9548A multiplexers (log but continue even if one fails)
75 err = i2c_init_tca9548(i2c_bus0_handle, &tca9548_bus0_handle);

83 }

89 }
90 else
91 {
92 ESP_LOGI(TAG, "TCA9548 initialized on bus 1 successfully");
93 }
94
95 // Main loop: read + average
96 while (1)
97 {
98 // DO NOT reinitialize sensors here — only read them
99 err = read_sensors_into_queue();
100 if (err != ESP_OK)
101 {
102 ESP_LOGE(TAG, "Failed to read sensors into queue: %s", esp_err_to_name(err));
103 }
104
105 sensor_data_average();
106
107 // Read sensors every 5 seconds
108 vTaskDelay(pdMS_TO_TICKS(5000));
109 }
