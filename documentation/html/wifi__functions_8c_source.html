26{

29 if (statemachine_mutex && xSemaphoreTake(statemachine_mutex, pdMS_TO_TICKS(10)) == pdTRUE)
30 {
31 sm_actuals.wifi_connected = false;
32 sm_actuals.wifi_error = false;
33 sm_actuals.wifi_ap_mode = false;
34 xSemaphoreGive(statemachine_mutex);
35 }
36
37 if (!s_wifi_initialized)

39 LOGI(TAG, "wifi_manager_stop: WiFi stop requested, but WiFi manager has never been initialized; no action taken");
40 update_syslog_state();

42 }
43
44 if (!s_wifi_initialized)
45 {
46 LOGI(TAG, "wifi_manager_stop: WiFi stop requested, but WiFi manager has never been initialized; no action taken");
47 return;
48 }
49
50 esp_err_t err = esp_wifi_disconnect();
51 if (err == ESP_ERR_WIFI_NOT_STARTED)
52 {
53 LOGI(TAG, "wifi_manager_stop: WiFi driver not started, skip disconnect");
54 }
55#if defined(ESP_ERR_WIFI_CONN_LOST)
56 else if (err != ESP_OK && err != ESP_ERR_WIFI_CONN_LOST)
57#else
58 else if (err != ESP_OK)
59#endif
60 {
61 LOGW(TAG, "esp_wifi_disconnect returned %s", esp_err_to_name(err));
62 }
63
64 err = esp_wifi_stop();
65 if (err == ESP_ERR_WIFI_NOT_STARTED)
66 {
67 LOGI(TAG, "wifi_manager_stop: WiFi driver not started, skip stop");
68 }
69 else if (err != ESP_OK)
70 {
71 LOGW(TAG, "esp_wifi_stop returned %s", esp_err_to_name(err));
72 }
73
74 if (s_wifi_event_group)
75 {
76 xEventGroupClearBits(s_wifi_event_group,
77 WIFI_CONNECTED_BIT | WIFI_FAIL_BIT | AP_STARTED_BIT);
78 }
79
80 if (statemachine_mutex && xSemaphoreTake(statemachine_mutex, pdMS_TO_TICKS(10)) == pdTRUE)
81 {
82 sm_actuals.wifi_connected = false;
83 sm_actuals.wifi_error = false;
84 sm_actuals.wifi_ap_mode = false;
85 xSemaphoreGive(statemachine_mutex);
86 }
87
88 update_syslog_state();

91esp_err_t wifi_manager_init(void)
92{
93 if (s_wifi_initialized)
94 {
95 LOGD(TAG, "wifi_manager_init: already initialized");
96 return ESP_OK;
97 }
98
99 if (s_wifi_event_group == NULL)
100 {
101 s_wifi_event_group = xEventGroupCreate();
102 if (s_wifi_event_group == NULL)
103 {
104 LOGE(TAG, "Failed to create WiFi event group");
105 return ESP_ERR_NO_MEM;
106 }
107 }
108
109 esp_err_t err = esp_netif_init();
110 if (err != ESP_OK && err != ESP_ERR_INVALID_STATE)
111 {
112 LOGE(TAG, "esp_netif_init failed: %s", esp_err_to_name(err));
113 return err;
114 }
115
116 esp_netif_t *existing_sta = esp_netif_get_handle_from_ifkey("WIFI_STA_DEF");
117 if (existing_sta == NULL)
118 {
119 esp_netif_t *netif = esp_netif_create_default_wifi_sta();
120 if (netif == NULL)
121 {
122 LOGE(TAG, "esp_netif_create_default_wifi_sta returned NULL");
123 return ESP_FAIL;
124 }
125 }
126 else
127 {
128 LOGI(TAG, "wifi_manager_init: reusing existing WIFI_STA_DEF netif %p", (void *)existing_sta);
129 }
130
131 if (network_config_mutex && xSemaphoreTake(network_config_mutex, pdMS_TO_TICKS(100)) == pdTRUE)
132 {
133 strncpy(hostname, wifi_config.hostname, sizeof(hostname) - 1);
134 xSemaphoreGive(network_config_mutex);
135 }
136 LOGI(TAG, "Setting WiFi hostname to: %s", hostname);
137 esp_netif_t *sta_netif = esp_netif_get_handle_from_ifkey("WIFI_STA_DEF");
138 if (sta_netif)
139 {
140 esp_err_t herr = esp_netif_set_hostname(sta_netif, hostname);
141 if (herr != ESP_OK)
142 LOGW(TAG, "esp_netif_set_hostname (STA) returned %s", esp_err_to_name(herr));
143 }
144
145 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
146 err = esp_wifi_init(&cfg);
147 if (err != ESP_OK && err != ESP_ERR_WIFI_NOT_INIT)
148 {
149 LOGE(TAG, "esp_wifi_init failed: %s", esp_err_to_name(err));
150 return err;
151 }
152
153 s_wifi_initialized = true;
154 LOGI(TAG, "WiFi manager initialized successfully");
155 return ESP_OK;
156}

158// select_wifi_mode: attempt STA only; do not perform automatic AP fallback.
159// network_manager_task will decide whether to call wifi_manager_start_ap().
160esp_err_t select_wifi_mode(void)
161{
162 const char *TAG_LOCAL = "WiFi";
163
164 /* If Ethernet is connected, do not attempt STA or AP selection here.
165 The network_manager will decide which action to take. */
166 if (eth_is_connected())
167 {
168 LOGW(TAG_LOCAL, "select_wifi_mode: Ethernet is up → skipping WiFi selection");
169 return ESP_OK;
170 }
171
172 /* Ensure WiFi is enabled in config */
173 if (strcmp(wifi_config.enable_wifi, "Off") == 0)
174 {
175 LOGW(TAG_LOCAL, "select_wifi_mode: WiFi disabled in config");
176 return ESP_ERR_INVALID_STATE;
177 }
178
179 /* If credentials are present, attempt STA only.
180 Do NOT automatically fall back to AP here — caller (network_manager) will decide. */
181 bool have_credentials =
182 strlen(wifi_config.ssid) > 0 &&
183 strlen(wifi_config.wifi_password) > 0;
184
185 if (!have_credentials)
186 {
187 LOGW(TAG_LOCAL, "select_wifi_mode: no WiFi credentials configured; caller may start AP");
188 return ESP_ERR_NOT_SUPPORTED;
189 }
190
191 LOGI(TAG_LOCAL, "select_wifi_mode: attempting WiFi STA");
192
193 esp_err_t r = wifi_manager_connect(&wifi_config);
194 if (r == ESP_OK)
195 {
196 LOGI(TAG_LOCAL, "select_wifi_mode: STA connected");
197 return ESP_OK;
198 }
199
200 /* STA failed — return failure to caller so it can decide whether to start AP */
201 LOGW(TAG_LOCAL, "select_wifi_mode: STA attempt failed (%s)", esp_err_to_name(r));
202 return r;
203}

205// Wifi manager connect and AP functions
206// Place these functions in wifi_functions.c (replace existing implementations)
207esp_err_t wifi_manager_connect(Wifi_config_t *settings)
208{
209 if (!settings)
210 {
211 LOGE(TAG, "wifi_manager_connect: settings pointer is NULL");
212 return ESP_ERR_INVALID_ARG;
213 }
214
215 if (network_config_mutex == NULL)
216 {
217 LOGE(TAG, "wifi_manager_connect: network_config_mutex is NULL");
218 return ESP_ERR_INVALID_STATE;
219 }
220
221 if (hw.has_ethernet)
222 {
223 esp_netif_t *eth = esp_netif_get_handle_from_ifkey("ETH_DEF");
224 if (eth)
225 {
226 esp_netif_ip_info_t ip;
227 if (esp_netif_get_ip_info(eth, &ip) == ESP_OK && ip.ip.addr != 0)
228 {
229 LOGI(TAG, "wifi_manager_connect: Ethernet has IP → skipping Wi‑Fi connect (Ethernet primary)");
230 return ESP_OK;
231 }
232 }
233
234 // Only defer Wi‑Fi while Ethernet is still actually alive and in a short startup window.
235 // Once the link is down / the interface has been torn down / unplugged, Wi‑Fi must be allowed to recover.
236 bool eth_is_live = (eth_handle != NULL && !eth_is_shutting_down() && ethernet_link_is_up());
237 if (eth_is_live)
238 {
239 LOGI(TAG, "wifi_manager_connect: Ethernet link up and DHCP pending → skipping Wi‑Fi connect for now");
240 return ESP_OK;
241 }
242
243 if (network_config_mutex && xSemaphoreTake(network_config_mutex, pdMS_TO_TICKS(50)) == pdTRUE)
244 {
245 bool eth_enabled = (strcmp(ethernet_config.enable_ethernet, "On") == 0);
246 xSemaphoreGive(network_config_mutex);
247 if (eth_enabled && eth_handle != NULL && !eth_is_shutting_down())
248 {
249 // Only defer for a short grace window while Ethernet is still coming up.
250 // Once the interface has been torn down or the link is absent, Wi‑Fi should take over.
251 if (ethernet_link_is_up())
252 {
253 LOGI(TAG, "wifi_manager_connect: Ethernet link is still live → deferring Wi‑Fi startup briefly");
254 return ESP_OK;
255 }
256 }
257 }
258 }
259
260 /* Copy settings under mutex to avoid holding it while blocking */
261 Wifi_config_t local;
262 memset(&local, 0, sizeof(local));
263 if (xSemaphoreTake(network_config_mutex, pdMS_TO_TICKS(100)) != pdTRUE)
264 {
265 LOGE(TAG, "wifi_manager_connect: failed to take network_config_mutex");
266 return ESP_ERR_TIMEOUT;
267 }
268 memcpy(&local, settings, sizeof(local));
269 xSemaphoreGive(network_config_mutex);
270
271 /* Ensure event group exists */
272 if (s_wifi_event_group == NULL)
273 {
274 LOGE(TAG, "wifi_manager_connect: s_wifi_event_group not initialized");
275 return ESP_ERR_INVALID_STATE;
276 }
277
278 /* Clear previous bits so wait is deterministic */
279 xEventGroupClearBits(s_wifi_event_group, WIFI_CONNECTED_BIT | WIFI_FAIL_BIT);
280
281 /* Prepare STA config */
282 wifi_config_t wifi_cfg = {0};
283 strncpy((char *)wifi_cfg.sta.ssid, local.ssid, sizeof(wifi_cfg.sta.ssid) - 1);
284 strncpy((char *)wifi_cfg.sta.password, local.wifi_password, sizeof(wifi_cfg.sta.password) - 1);
285 wifi_cfg.sta.threshold.authmode = WIFI_AUTH_WPA2_PSK;
286 wifi_cfg.sta.sae_pwe_h2e = WPA3_SAE_PWE_BOTH;
287 wifi_cfg.sta.pmf_cfg.capable = true;
288 wifi_cfg.sta.pmf_cfg.required = false;
289
290 /* Ensure STA netif exists (create if missing) */
291 esp_netif_t *sta_netif = esp_netif_get_handle_from_ifkey("WIFI_STA_DEF");
292 if (!sta_netif)
293 {
294 sta_netif = esp_netif_create_default_wifi_sta();
295 if (!sta_netif)
296 {
297 LOGE(TAG, "wifi_manager_connect: WIFI_STA_DEF netif not found and could not be created");
298 return ESP_ERR_NOT_FOUND;
299 }
300 }
301
302 /* Configure static IP if requested (apply before start) */
303 if (strcmp(local.enable_dhcp, "Off") == 0)
304 {
305 esp_err_t r = esp_netif_dhcpc_stop(sta_netif);
306 if (r != ESP_OK && r != ESP_ERR_ESP_NETIF_DHCP_ALREADY_STOPPED)
307 LOGW(TAG, "wifi_manager_connect: dhcpc_stop -> %s", esp_err_to_name(r));
308
309 esp_netif_ip_info_t ip_info = {0};
310 if (inet_pton(AF_INET, local.ip_address, &ip_info.ip) != 1 ||
311 inet_pton(AF_INET, local.subnet_mask, &ip_info.netmask) != 1 ||
312 inet_pton(AF_INET, local.gateway, &ip_info.gw) != 1)
313 {
314 LOGE(TAG, "wifi_manager_connect: invalid static IP fields");
315 return ESP_ERR_INVALID_ARG;
316 }
317
318 if (esp_netif_set_ip_info(sta_netif, &ip_info) != ESP_OK)
319 {
320 LOGE(TAG, "wifi_manager_connect: esp_netif_set_ip_info failed");
321 return ESP_FAIL;
322 }
323
324 esp_netif_dns_info_t dns_info = {0};
325 if (inet_pton(AF_INET, local.primary_dns, &dns_info.ip.u_addr.ip4) == 1)
326 {
327 dns_info.ip.type = ESP_IPADDR_TYPE_V4;
328 esp_netif_set_dns_info(sta_netif, ESP_NETIF_DNS_MAIN, &dns_info);
329 }
330 if (strlen(local.secondary_dns) > 0 && inet_pton(AF_INET, local.secondary_dns, &dns_info.ip.u_addr.ip4) == 1)
331 {
332 dns_info.ip.type = ESP_IPADDR_TYPE_V4;
333 esp_netif_set_dns_info(sta_netif, ESP_NETIF_DNS_BACKUP, &dns_info);
334 }
335 }
336 else
337 {
338 esp_err_t r = esp_netif_dhcpc_start(sta_netif);
339 if (r != ESP_OK && r != ESP_ERR_ESP_NETIF_DHCP_ALREADY_STARTED)
340 LOGW(TAG, "wifi_manager_connect: dhcpc_start -> %s", esp_err_to_name(r));
341 }
342
343 /* Set mode and config */
344 esp_err_t err = esp_wifi_set_mode(WIFI_MODE_STA);
345 if (err != ESP_OK)
346 {
347 LOGE(TAG, "wifi_manager_connect: esp_wifi_set_mode -> %s", esp_err_to_name(err));
348 return err;
349 }
350
351 err = esp_wifi_set_config(WIFI_IF_STA, &wifi_cfg);
352 if (err != ESP_OK)
353 {
354 LOGE(TAG, "wifi_manager_connect: esp_wifi_set_config -> %s", esp_err_to_name(err));
355 return err;
356 }
357
358 /* Start Wi‑Fi (idempotent) */
359 err = esp_wifi_start();
360 if (err != ESP_OK && err != ESP_ERR_WIFI_NOT_STARTED)
361 {
362 LOGE(TAG, "wifi_manager_connect: esp_wifi_start -> %s", esp_err_to_name(err));
363 return err;
364 }
365
366 /* Trigger connect (may be no-op if already connecting) */
367 esp_err_t conn_err = esp_wifi_connect();
368 if (conn_err != ESP_OK && conn_err != ESP_ERR_WIFI_NOT_STARTED)
369 {
370 LOGW(TAG, "wifi_manager_connect: esp_wifi_connect -> %s", esp_err_to_name(conn_err));
371 }
372
373 /* Wait for driver-level connected event first (10 s) */
374 const TickType_t connect_wait = pdMS_TO_TICKS(10000);
375 EventBits_t bits = xEventGroupWaitBits(
376 s_wifi_event_group,
377 WIFI_CONNECTED_BIT | WIFI_FAIL_BIT,
378 pdTRUE, /* clear bits on exit */
379 pdFALSE, /* wait for any */
380 connect_wait);
381
382 if (bits & WIFI_FAIL_BIT)
383 {
384 LOGW(TAG, "wifi_manager_connect: WIFI_FAIL_BIT set");
385 return ESP_FAIL;
386 }
387
388 /* Poll for a valid IP (allow DHCP time) */
389 const TickType_t ip_poll_total = pdMS_TO_TICKS(20000);
390 const TickType_t ip_poll_interval = pdMS_TO_TICKS(500);
391 TickType_t waited = 0;
392 while (waited < ip_poll_total)
393 {
394 esp_netif_ip_info_t ipinfo;
395 if (esp_netif_get_ip_info(sta_netif, &ipinfo) == ESP_OK && ipinfo.ip.addr != 0)
396 {
397 extern int s_retry_num;
398 s_retry_num = 0;
399 LOGI(TAG, "wifi_manager_connect: IP assigned: " IPSTR, IP2STR(&ipinfo.ip));
400 return ESP_OK;
401 }
402 vTaskDelay(ip_poll_interval);
403 waited += ip_poll_interval;
404 }
405
406 LOGW(TAG, "wifi_manager_connect: connection timeout to SSID '%s' (no IP after %d ms)", local.ssid, (int)pdTICKS_TO_MS(ip_poll_total));
407 return ESP_ERR_TIMEOUT;
408}

410// wifi_manager_start_ap: start AP mode and return status. Caller must decide when to call this.
411esp_err_t wifi_manager_start_ap(void)
412{
413 if (xSemaphoreTake(network_config_mutex, pdMS_TO_TICKS(100)) != pdTRUE)
414 {
415 LOGE(TAG, "wifi_manager_start_ap: failed to take network_config_mutex");
416 return ESP_ERR_TIMEOUT;
417 }
418
419 /* Create or reuse AP netif */
420 esp_netif_t *ap_existing = esp_netif_get_handle_from_ifkey("WIFI_AP_DEF");
421 if (ap_existing == NULL)
422 {
423 esp_netif_t *ap_netif = esp_netif_create_default_wifi_ap();
424 if (!ap_netif)
425 {
426 LOGE(TAG, "wifi_manager_start_ap: esp_netif_create_default_wifi_ap returned NULL");
427 xSemaphoreGive(network_config_mutex);
428 return ESP_FAIL;
429 }
430 }
431 else
432 {
433 LOGI(TAG, "wifi_manager_start_ap: reusing existing WIFI_AP_DEF netif %p", (void *)ap_existing);
434 }
435
436 wifi_config_t wifi_cfg = {
437 .ap = {
438 .ssid = ESP_AP_SSID,
439 .ssid_len = strlen(ESP_AP_SSID),
440 .channel = ESP_AP_CHANNEL,
441 .password = ESP_AP_PASS,
442 .max_connection = ESP_AP_MAX_CONN,
443 .authmode = strlen(ESP_AP_PASS) > 0 ? WIFI_AUTH_WPA2_PSK : WIFI_AUTH_OPEN,
444 .pmf_cfg = {
445 .required = false,
446 },
447 },
448 };
449
450 esp_err_t err = esp_wifi_set_mode(WIFI_MODE_AP);
451 if (err != ESP_OK)
452 {
453 LOGE(TAG, "wifi_manager_start_ap: esp_wifi_set_mode(AP) failed: %s", esp_err_to_name(err));
454 xSemaphoreGive(network_config_mutex);
455 return err;
456 }
457
458 err = esp_wifi_set_config(WIFI_IF_AP, &wifi_cfg);
459 if (err != ESP_OK)
460 {
461 LOGE(TAG, "wifi_manager_start_ap: esp_wifi_set_config(AP) failed: %s", esp_err_to_name(err));
462 xSemaphoreGive(network_config_mutex);
463 return err;
464 }
465
466 err = esp_wifi_start();
467 if (err != ESP_OK)
468 {
469 LOGE(TAG, "wifi_manager_start_ap: esp_wifi_start(AP) failed: %s", esp_err_to_name(err));
470 xSemaphoreGive(network_config_mutex);
471 return err;
472 }
473
474 /* Release mutex before waiting */
475 xSemaphoreGive(network_config_mutex);
476
477 /* Wait for AP_STARTED_BIT set by event handler */
478 EventBits_t bits = xEventGroupWaitBits(
479 s_wifi_event_group,
480 AP_STARTED_BIT,
481 pdFALSE, pdFALSE,
482 pdMS_TO_TICKS(5000));
483
484 if (bits & AP_STARTED_BIT)
485 {
486 LOGI(TAG, "wifi_manager_start_ap: AP started successfully (SSID=%s)", ESP_AP_SSID);
487 return ESP_OK;
488 }
489
490 LOGE(TAG, "wifi_manager_start_ap: AP failed to start (timeout)");
491 return ESP_FAIL;
492}
