36 LOGE(TAG, "Failed to initialize DS3231 on bus 0: %s", esp_err_to_name(err));

43 while (1)

60 LOGI(TAG, "DS3231 successfully synced with NTP");
61 }
62 else
63 {
64 LOGW(TAG, "Sync attempt failed: %s", esp_err_to_name(err));
65 }
66
67 // Sleep for one hour before next sync
