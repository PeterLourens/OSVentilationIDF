21void stop_task_i2c_lcd(void)
22{
23 if (task_i2c_lcd == NULL)
24 {
25 LOGI(TAG, "LCD task already stopped");
26 return;
27 }
28
29 // Optional: turn off LCD hardware
30 i2c_lcd_clear();
31 i2c_lcd_backlight_set(false);
32
33 LOGI(TAG, "Stopping LCD task...");
34 vTaskDelete(task_i2c_lcd);
35 task_i2c_lcd = NULL;

38void start_task_i2c_lcd(void)
39{
40 I2C_config_t cfg;
41
42 xSemaphoreTake(i2c_config_mutex, portMAX_DELAY);
43 memcpy(&cfg, &i2c_config, sizeof(cfg));
44 xSemaphoreGive(i2c_config_mutex);
45
46 if (strcmp(cfg.enable_lcd, "On") != 0)
47 {
48 LOGI(TAG, "LCD disabled → task not started");
49 return;
50 }
51
52 if (task_i2c_lcd != NULL)
53 {
54 LOGW(TAG, "LCD task already running");
55 return;
56 }
57
58 task_i2c_lcd = xTaskCreateStatic(task_i2c_lcd_code, "i2c_lcd", i2c_lcd_stack_size, NULL, 6, i2c_lcd_stack, &i2c_lcd_tcb);
59
60 if (!task_i2c_lcd)
61 LOGE(TAG, "LCD task creation failed");
62 else
63 LOGI(TAG, "LCD task started");
64}

66void task_i2c_lcd_code(void *pvParameters)
67{
68 I2C_config_t cfg;
69 char uptime[20] = {};
70 char ping_result[20] = {};
71 char formatted_time[32] = {};
72 char wifi_ip[16] = "0.0.0.0";
73 char eth_ip[16] = "0.0.0.0";
74 char buf[200] = {};
75
76 // Copy config under mutex if available
77 if (i2c_config_mutex && xSemaphoreTake(i2c_config_mutex, portMAX_DELAY) == pdTRUE)
78 {
79 memcpy(&cfg, &i2c_config, sizeof(cfg));
80 xSemaphoreGive(i2c_config_mutex);
81 }
82 else
83 {
84 LOGW(TAG, "i2c_config_mutex not available; using defaults");
85 memset(&cfg, 0, sizeof(cfg));
86 }
87
88 if (strcmp(cfg.enable_lcd, "Off") == 0)
89 {
90 LOGW(TAG, "LCD disabled → terminating LCD task");
91 task_i2c_lcd = NULL;
92 vTaskDelete(NULL);
93 }
94
95 // LCD availability flag (degraded mode)
96 bool lcd_available = false;
97
98 // Try to initialize LCD device with retries and backoff.
99 const TickType_t retry_initial_delay = pdMS_TO_TICKS(1000);
100 const TickType_t retry_long_delay = pdMS_TO_TICKS(30000);
101
102 for (;;)
103 {
104 if (!lcd_available)
105 {
106 if (i2c_lcd_init_device() == ESP_OK)
107 {
108 lcd_available = true;
109 LOGI(TAG, "LCD device initialized and available");
110 // Ensure display is in known state
111 i2c_lcd_clear();
112 i2c_lcd_backlight_set(false);
113 }
114 else
115 {
116 LOGW(TAG, "LCD init failed; will retry in 1s");
117 vTaskDelay(retry_initial_delay);
118 // Try again a few times quickly, then back off
119 if (i2c_lcd_init_device() != ESP_OK)
120 {
121 LOGW(TAG, "LCD init retry failed; backing off 30s");
122 vTaskDelay(retry_long_delay);
123 continue;
124 }
125 else
126 {
127 lcd_available = true;
128 LOGI(TAG, "LCD device initialized on retry");
129 i2c_lcd_clear();
130 i2c_lcd_backlight_set(false);
131 }
132 }
133 }
134
135 // If LCD is available, perform normal update cycle
136 if (lcd_available)
137 {
138 // Display 1 - General info
139 formatted_uptime(uptime, sizeof(uptime));
140 formatted_datetime(formatted_time, sizeof(formatted_time));
141 snprintf(buf, sizeof(buf), "%s\nUp: %s", formatted_time, uptime);
142 if (i2c_lcd_write_text(buf) != ESP_OK)
143 {
144 LOGW(TAG, "Failed to write general info to LCD; marking LCD unavailable");
145 lcd_available = false;
146 // small delay before next init attempt
147 vTaskDelay(pdMS_TO_TICKS(500));
148 continue;
149 }
150 vTaskDelay(pdMS_TO_TICKS(3000));
151
152 // Display 2 - Network info
153 if (statemachine_mutex && xSemaphoreTake(statemachine_mutex, pdMS_TO_TICKS(20)) == pdTRUE)
154 {
155 if (sm_actuals.wifi_ip_addr[0] != '\0')
156 strncpy(wifi_ip, sm_actuals.wifi_ip_addr, sizeof(wifi_ip) - 1);
157 if (sm_actuals.eth_ip_addr[0] != '\0' && hw.has_ethernet)
158 strncpy(eth_ip, sm_actuals.eth_ip_addr, sizeof(eth_ip) - 1);
159 xSemaphoreGive(statemachine_mutex);
160 }
161 else
162 {
163 LOGD(TAG, "statemachine_mutex not available for network info");
164 }
165
166 if (ping_gateway())
167 {
168 snprintf(ping_result, sizeof(ping_result), "Gateway OK");
169 }
170 else
171 {
172 snprintf(ping_result, sizeof(ping_result), "Gateway unreachable");
173 }
174
175 snprintf(buf, sizeof(buf), "Wifi:%s\nEth:%s\n%s", wifi_ip, eth_ip, ping_result);
176 if (i2c_lcd_write_text(buf) != ESP_OK)
177 {
178 LOGW(TAG, "Failed to write network info to LCD; marking LCD unavailable");
179 lcd_available = false;
180 vTaskDelay(pdMS_TO_TICKS(500));
181 continue;
182 }
183 vTaskDelay(pdMS_TO_TICKS(3000));
184
185 // Display 3 - Statemachine state
186 build_sm_actuals_buffer(buf, sizeof(buf));
187 if (i2c_lcd_write_text(buf) != ESP_OK)
188 {
189 LOGW(TAG, "Failed to write state info to LCD; marking LCD unavailable");
190 lcd_available = false;
191 vTaskDelay(pdMS_TO_TICKS(500));
192 continue;
193 }
194 vTaskDelay(pdMS_TO_TICKS(3000));
195
196 // Display 4 - Sensors
197 // Wait for sensor queues to be ready
198 while (!sensors_queues_ready)
199 {
200 LOGD(TAG, "Waiting for sensor queues to be ready...");
201 vTaskDelay(pdMS_TO_TICKS(50));
202 }
203
204 if (sensor_avg_item_size != sizeof(float[MAX_I2C_BUSES][CHANNELS_PER_BUS][DATA_FIELDS]))
205 {
206 LOGE(TAG, "sensor_avg_queue item size mismatch: expected %zu got %zu",
207 sizeof(float[MAX_I2C_BUSES][CHANNELS_PER_BUS][DATA_FIELDS]), sensor_avg_item_size);
208 // skip sensor display but keep LCD alive
209 }
210 else
211 {
212 float avg[MAX_I2C_BUSES][CHANNELS_PER_BUS][DATA_FIELDS] = {0};
213 if (xQueuePeek(sensor_avg_queue, avg, 0) != pdTRUE)
214 {
215 LOGW(TAG, "Failed to read avg sensor queue; skipping sensor display");
216 }
217 else
218 {
219 for (int bus = 0; bus < MAX_I2C_BUSES; bus++)
220 {
221 for (int ch = 0; ch < CHANNELS_PER_BUS; ch++)
222 {
223 bool active = false;
224 if (sensor_cfg_mutex && xSemaphoreTake(sensor_cfg_mutex, pdMS_TO_TICKS(20)) == pdTRUE)
225 {
226 const bus_sensor_t *meta = (bus == 0) ? &bus0_sensor_settings[ch] : &bus1_sensor_settings[ch];
227 active = (strcmp(meta->type, "None") != 0);
228 xSemaphoreGive(sensor_cfg_mutex);
229 }
230 else
231 {
232 LOGD(TAG, "sensor_cfg_mutex not available; assuming sensor active");
233 active = true;
234 }
235
236 if (!active)
237 continue;
238
239 format_sensor_overview(buf, sizeof(buf), bus, ch, avg);
240 if (i2c_lcd_write_text(buf) != ESP_OK)
241 {
242 LOGW(TAG, "Failed to write sensor overview for b%ds%d; marking LCD unavailable", bus, ch);
243 lcd_available = false;
244 break;
245 }
246 vTaskDelay(pdMS_TO_TICKS(3000));
247 }
248 if (!lcd_available)
249 break;
250 }
251 }
252 }
253
254 if (!lcd_available)
255 {
256 // short delay before re-init attempts
257 vTaskDelay(pdMS_TO_TICKS(500));
258 continue;
259 }
260
261 // Display 5 - Valve positions
262 build_valve_positions_buffer(buf, sizeof(buf));
263 if (i2c_lcd_write_text(buf) != ESP_OK)
264 {
265 LOGW(TAG, "Failed to write valve positions to LCD; marking LCD unavailable");
266 lcd_available = false;
267 vTaskDelay(pdMS_TO_TICKS(500));
268 continue;
269 }
270 vTaskDelay(pdMS_TO_TICKS(3000));
271 }
272 else
273 {
274 // LCD not available: wait and retry init periodically
275 LOGI(TAG, "LCD unavailable; retrying initialization in 30s");
276 vTaskDelay(retry_long_delay);
277 // next loop iteration will attempt init
278 }
279 }
