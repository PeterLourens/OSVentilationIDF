31void task_influxdb_code(void *pvParameters)
32{
33 (void)pvParameters;
34
35 // initial startup delay
36 vTaskDelay(pdMS_TO_TICKS(5000));
37
38 while (!network_has_ip())
39 {
40 LOGW(TAG, "InfluxDB: Waiting for network...");
41 vTaskDelay(pdMS_TO_TICKS(2000));
42 }
43
44 LOGI(TAG, "InfluxDB: Network ready");
45
46 if (!influxdb_wait_ready(pdMS_TO_TICKS(2000)))
47 {
48 LOGW(TAG, "Influx sender not ready after wait; enqueues will be skipped until ready");
49 }
50 else
51 {
52 LOGI(TAG, "Influx sender ready (queue=%p)", (void *)influx_queue);
53 }
54
55 // Use local copies of intervals so reads are atomic-ish
56 uint32_t influx_interval_ms = g_influx_task_interval_ms;
57 uint32_t monitor_interval_ms = g_dma_monitor_interval_ms;
58 uint32_t tick_resolution_ms = g_monitor_tick_resolution_ms;
59
60 // Track elapsed time (ms)
61 uint32_t elapsed_influx_ms = 0;
62 uint32_t elapsed_monitor_ms = 0;
63
64 // Ensure resolution is sane
65 if (tick_resolution_ms == 0)
66 tick_resolution_ms = 1000;
67
68 // Run once immediately (optional)
69 LOGI(TAG, "Update InfluxDB (initial)...");
70 influxdb_process(sensor_queue, sensor_avg_queue);
71 // After initial publish, reset counters
72 elapsed_influx_ms = 0;
73 elapsed_monitor_ms = 0;
74
75 for (;;)
76 {
77 // Sleep for the resolution slice
78 vTaskDelay(pdMS_TO_TICKS(tick_resolution_ms));
79
80 // Update intervals in case they were changed at runtime
81 influx_interval_ms = g_influx_task_interval_ms;
82 monitor_interval_ms = g_dma_monitor_interval_ms;
83
84 // Increment elapsed counters
85 elapsed_influx_ms += tick_resolution_ms;
86 elapsed_monitor_ms += tick_resolution_ms;
87
88 // If monitor interval reached, enqueue DMA metric
89 if (elapsed_monitor_ms >= monitor_interval_ms)
90 {
91 // Enqueue DMA metrics (non-blocking inside)
92 if (!influxdb_enqueue_dma_metrics("osvent"))
93 {
94 LOGW(TAG, "influxdb: dma monitor enqueue failed");
95 }
96 // reset monitor counter but keep any overshoot
97 elapsed_monitor_ms = elapsed_monitor_ms % monitor_interval_ms;
98 }
99
100 // If influx interval reached, run the full Influx process
101 if (elapsed_influx_ms >= influx_interval_ms)
102 {
103 LOGI(TAG, "Update InfluxDB...");
104 influxdb_process(sensor_queue, sensor_avg_queue);
105
106 // Reset influx counter but keep overshoot
107 elapsed_influx_ms = elapsed_influx_ms % influx_interval_ms;
108 }
109 }
110}
