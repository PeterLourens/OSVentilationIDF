37static int bind_socket_to_active_interface(int sock)
38{
39 esp_netif_t *netif = get_active_netif();
40 if (!netif)
41 {
42 ESP_LOGW(TAG, "bind_socket_to_active_interface: no active network interface available");
43 return -1;
44 }
45
46 esp_netif_ip_info_t ipinfo;
47 if (esp_netif_get_ip_info(netif, &ipinfo) != ESP_OK)
48 {
49 ESP_LOGW(TAG, "bind_socket_to_active_interface: failed to get ip info");
50 return -1;
51 }
52
53 if (ipinfo.ip.addr == 0)
54 {
55 ESP_LOGW(TAG, "bind_socket_to_active_interface: active netif has no IP");
56 return -1;
57 }
58
59 struct sockaddr_in local = {0};
60 local.sin_family = AF_INET;
61 local.sin_port = htons(0);
62 local.sin_addr.s_addr = ipinfo.ip.addr;
63
64 if (bind(sock, (struct sockaddr *)&local, sizeof(local)) != 0)
65 {
66 ESP_LOGW(TAG, "bind_socket_to_active_interface: bind failed: errno=%d (%s)", errno, strerror(errno));
67 return -1;
68 }
69
70 ESP_LOGD(TAG, "bind_socket_to_active_interface: bound to %s", inet_ntoa(*(struct in_addr *)&ipinfo.ip.addr));
71 return 0;
72}

74static ssize_t safe_syslog_send(int sockfd, const void *buf, size_t len, int flags, const struct sockaddr *dest, socklen_t destlen)
75{
76 if (sockfd < 0)
77 {
78 ESP_LOGD(TAG, "safe_syslog_send: invalid socket fd=%d", sockfd);
79 return -1;
80 }
81
82 int64_t now_ms = esp_timer_get_time() / 1000LL;
83 if (s_last_syslog_fail_ms != 0 &&
84 (now_ms - s_last_syslog_fail_ms) < SYSLOG_RETRY_BACKOFF_MS)
85 {
86 ESP_LOGD(TAG, "safe_syslog_send: skipping send due to backoff (elapsed=%lldms)", (long long)(now_ms - s_last_syslog_fail_ms));
87 return -1;
88 }
89
90 ssize_t r = sendto(sockfd, buf, len, flags | MSG_DONTWAIT, dest, destlen);
91 if (r >= 0)
92 {
93 s_last_syslog_fail_ms = 0;
94 return r;
95 }
96
97 int e = errno;
98 ESP_LOGW(TAG, "syslog sendto failed: fd=%d errno=%d (%s)", sockfd, e, strerror(e));
99
100 // Treat common transient errors specially; apply backoff but keep socket open.
101 if (e == EAGAIN || e == EWOULDBLOCK || e == EINTR)
102 {
103 s_last_syslog_fail_ms = now_ms;
104 return -1;
105 }
106
107 // For other errors, apply backoff and return failure. The caller (which creates and closes the socket per message) will close the socket.
108 s_last_syslog_fail_ms = now_ms;
109 return -1;
110}

112static void syslog_worker_task(void *pvParameters)
113{
114 (void)pvParameters;
115 syslog_queue_item_t item;
116
117 for (;;)
118 {
119 static bool syslog_running = false;
120
121 // --- ASYNCHRONOUS SYSLOG STATE UPDATE ---
122 if (g_request_syslog_update)
123 {
124 g_request_syslog_update = false;
125
126 bool ready = network_has_ip();
127
128 bool enabled_cfg = false;
129 if (syslog_config_mutex &&
130 xSemaphoreTake(syslog_config_mutex, pdMS_TO_TICKS(20)) == pdTRUE)
131 {
132 enabled_cfg = (strcmp(syslog_config.enable_syslog, "On") == 0);
133 xSemaphoreGive(syslog_config_mutex);
134 }
135
136 if (enabled_cfg && ready && !syslog_running)
137 {
138 syslog_enable();
139 syslog_running = true;
140 ESP_LOGI("syslog", "Syslog enabled");
141 }
142 else if ((!enabled_cfg || !ready) && syslog_running)
143 {
144 syslog_disable();
145 syslog_running = false;
146 ESP_LOGI("syslog", "Syslog disabled");
147 }
148 }
149
150 if (s_syslog_queue == NULL)
151 {
152 vTaskDelay(pdMS_TO_TICKS(1000));
153 continue;
154 }
155
156 if (xQueueReceive(s_syslog_queue, &item, portMAX_DELAY) != pdTRUE)
157 continue;
158
159 if (!item.line)
160 continue;
161
162 /* Snapshot config */
163 char ip[MEDIUM_CONFIG_ITEM] = {0};
164 char port_str[SMALL_CONFIG_ITEM] = {0};
165 char hostname_local[MEDIUM_CONFIG_ITEM] = {0};
166 char appname_local[MEDIUM_CONFIG_ITEM] = {0};
167 bool enabled = false;
168
169 if (syslog_config_mutex && xSemaphoreTake(syslog_config_mutex, pdMS_TO_TICKS(50)) == pdTRUE)
170 {
171 enabled = (strcmp(syslog_config.enable_syslog, "On") == 0);
172 if (enabled)
173 {
174 strncpy(ip, syslog_config.syslog_ip, sizeof(ip) - 1);
175 strncpy(port_str, syslog_config.syslog_port, sizeof(port_str) - 1);
176 strncpy(hostname_local, syslog_config.syslog_hostname, sizeof(hostname_local) - 1);
177 strncpy(appname_local, syslog_config.syslog_appname, sizeof(appname_local) - 1);
178 }
179 xSemaphoreGive(syslog_config_mutex);
180 }
181 else
182 {
183 ESP_LOGW(TAG, "syslog_worker: could not take syslog_config_mutex; skipping send");
184 heap_caps_free(item.line);
185 continue;
186 }
187
188 if (!enabled)
189 {
190 heap_caps_free(item.line);
191 continue;
192 }
193
194 int port = atoi(port_str);
195 if (port <= 0 || port > 65535)
196 {
197 ESP_LOGE(TAG, "syslog_worker: invalid port '%s', dropping message", port_str);
198 heap_caps_free(item.line);
199 continue;
200 }
201
202 if (strlen(ip) == 0)
203 {
204 ESP_LOGE(TAG, "syslog_worker: syslog IP empty, dropping message");
205 heap_caps_free(item.line);
206 continue;
207 }
208
209 // Build syslog message
210 char *msg = NULL;
211 int n = asprintf(&msg, "<13>%s %s: %s\n", hostname_local[0] ? hostname_local : "set-hostname", appname_local[0] ? appname_local : "set-app", item.line);
212 if (n < 0 || !msg)
213 {
214 ESP_LOGE(TAG, "syslog_worker: asprintf failed");
215 heap_caps_free(item.line);
216 continue;
217 }
218
219 // Prepare destination address
220 struct sockaddr_in dest;
221 memset(&dest, 0, sizeof(dest));
222 dest.sin_family = AF_INET;
223 dest.sin_port = htons(port);
224 if (!inet_aton(ip, &dest.sin_addr))
225 {
226 ESP_LOGE(TAG, "syslog_worker: inet_aton failed for %s", ip);
227 heap_caps_free(msg);
228 heap_caps_free(item.line);
229 continue;
230 }
231
232 while (!network_has_ip())
233 {
234 ESP_LOGW(TAG, "syslog_worker: no active network interface; waiting for IP before sending syslog");
235 if (!network_wait_for_ip(10, 500, NULL))
236 {
237 ESP_LOGW(TAG, "syslog_worker: still no IP available; retrying");
238 }
239 }
240
241 // Create UDP socket
242 int sock = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
243 if (sock < 0)
244 {
245 ESP_LOGE(TAG, "syslog_worker: socket() failed: errno=%d (%s)", errno, strerror(errno));
246 heap_caps_free(msg);
247 heap_caps_free(item.line);
248 vTaskDelay(pdMS_TO_TICKS(SYSLOG_RETRY_DELAY_MS));
249 continue;
250 }
251
252 // Bind socket to active interface IP so outgoing packets use that interface
253 if (bind_socket_to_active_interface(sock) != 0)
254 {
255 ESP_LOGW(TAG, "syslog_worker: failed to bind socket to active interface; continuing (socket will use default route)");
256 // Not fatal: continue and attempt send
257 }
258
259 // Try to send with a few retries for transient errors
260 ssize_t sent = -1;
261 for (int attempt = 0; attempt < SYSLOG_SEND_RETRIES; ++attempt)
262 {
263 sent = safe_syslog_send(sock, msg, strlen(msg), 0, (struct sockaddr *)&dest, sizeof(dest));
264 if (sent >= 0)
265 break;
266 vTaskDelay(pdMS_TO_TICKS(SYSLOG_RETRY_DELAY_MS));
267 }
268
269 if (sent < 0)
270 {
271 ESP_LOGE(TAG, "syslog_worker: send failed after %d attempts", SYSLOG_SEND_RETRIES);
272 }
273 else
274 {
275 ESP_LOGD(TAG, "syslog_worker: sent %d bytes", (int)sent);
276 // Notify network manager that Ethernet activity occurred
277 net_mgr_notify_eth_activity();
278 }
279
280 close(sock);
281 heap_caps_free(msg);
282 heap_caps_free(item.line);
283 }
284}

334void syslog_enqueue_line(const char *line)
335{
336 if (!line || !*line)
337 return;
338
339 if (s_syslog_queue == NULL)
340 {
341 // Queue not initialized: fall back to synchronous send
342 ESP_LOGW(TAG, "syslog_enqueue_line: queue not initialized, falling back to direct send");
343 // call original synchronous implementation below
344 syslog_send_line(line);
345 return;
346 }
347
348 size_t len = strnlen(line, SYSLOG_MAX_LINE_LEN);
349 char *copy = heap_caps_malloc(len + 1, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
350 if (!copy)
351 {
352 ESP_LOGE(TAG, "syslog_enqueue_line: heap_caps_malloc failed");
353 return;
354 }
355 memcpy(copy, line, len);
356 copy[len] = '\0';
357
358 syslog_queue_item_t item = {.line = copy, .len = len};
359
360 if (xQueueSend(s_syslog_queue, &item, 0) != pdTRUE)
361 {
362 ESP_LOGW(TAG, "syslog_enqueue_line: queue full, dropping message");
363 heap_caps_free(copy);
364 }
365}

367void syslog_send_line(const char *line)
368{
369 // If queue exists, enqueue and return quickly
370 if (s_syslog_queue != NULL)
371 {
372 syslog_enqueue_line(line);
373 return;
374 }
375
376 if (!network_has_ip())
377 {
378 LOGI(TAG, "syslog_send_line: network not ready, skipping send");
379 return;
380 }
381
382 if (!line || !*line)
383 {
384 LOGW(TAG, "syslog_send_line: empty line, nothing to send");
385 return;
386 }
387
388 // Read syslog config safely
389 char ip[MEDIUM_CONFIG_ITEM] = {0};
390 char port_str[SMALL_CONFIG_ITEM] = {0};
391 char hostname_local[MEDIUM_CONFIG_ITEM] = {0};
392 char appname_local[MEDIUM_CONFIG_ITEM] = {0};
393 char device_id_local[MEDIUM_CONFIG_ITEM] = {0};
394
395 bool enabled = false;
396
397 xSemaphoreTake(syslog_config_mutex, portMAX_DELAY);
398
399 enabled = (strcmp(syslog_config.enable_syslog, "On") == 0);
400
401 if (enabled)
402 {
403 strncpy(ip, syslog_config.syslog_ip, sizeof(ip) - 1);
404 strncpy(port_str, syslog_config.syslog_port, sizeof(port_str) - 1);
405 strncpy(hostname_local, syslog_config.syslog_hostname, sizeof(hostname_local) - 1);
406 strncpy(appname_local, syslog_config.syslog_appname, sizeof(appname_local) - 1);
407 strncpy(device_id_local, syslog_config.syslog_device_id, sizeof(device_id_local) - 1);
408 }
409
410 xSemaphoreGive(syslog_config_mutex);
411
412 if (!enabled)
413 return;
414
415 int port = atoi(port_str);
416 if (port <= 0 || port > 65535)
417 {
418 LOGE(TAG, "Invalid syslog port: %s", port_str);
419 return;
420 }
421
422 if (strlen(ip) == 0)
423 {
424 LOGE(TAG, "Syslog IP not configured");
425 return;
426 }
427
428 if (strlen(hostname_local) == 0)
429 snprintf(hostname_local, sizeof(hostname_local), "%s", "set hostname");
430
431 if (strlen(appname_local) == 0)
432 snprintf(appname_local, sizeof(appname_local), "%s", "set app name");
433
434 if (strlen(device_id_local) == 0)
435 snprintf(device_id_local, sizeof(device_id_local), "%s", "set device ID");
436
437 size_t msg_len = strlen(hostname_local) + strlen(appname_local) + strlen(line) + 64;
438 char *msg = heap_caps_malloc(msg_len, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
439 if (!msg)
440 {
441 LOGE(TAG, "syslog_send_line: heap_caps_malloc failed for %zu bytes", msg_len);
442 return;
443 }
444
445 int n = snprintf(msg, msg_len, "<13>%s %s: %s\n", hostname_local, appname_local, line);
446 if (n < 0 || (size_t)n >= msg_len)
447 {
448 LOGE(TAG, "syslog_send_line: snprintf failed or truncated");
449 heap_caps_free(msg);
450 return;
451 }
452
453 int sock = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
454 if (sock < 0)
455 {
456 ESP_LOGE(TAG, "socket() failed: errno=%d (%s)", errno, strerror(errno));
457 heap_caps_free(msg);
458 return;
459 }
460
461 // Bind to active interface if possible
462 if (bind_socket_to_active_interface(sock) != 0)
463 {
464 ESP_LOGW(TAG, "syslog_send_line: failed to bind socket to active interface; continuing");
465 }
466
467 struct sockaddr_in dest;
468 memset(&dest, 0, sizeof(dest));
469 dest.sin_family = AF_INET;
470 dest.sin_port = htons(port);
471
472 if (!inet_aton(ip, &dest.sin_addr))
473 {
474 LOGE(TAG, "inet_aton failed for %s", ip);
475 close(sock);
476 heap_caps_free(msg);
477 return;
478 }
479
480 ssize_t sent = safe_syslog_send(sock, msg, strlen(msg), 0, (struct sockaddr *)&dest, sizeof(dest));
481
482 if (sent < 0)
483 {
484 LOGE(TAG, "syslog: send failed (see earlier diagnostics)");
485 }
486 else
487 {
488 LOGD(TAG, "Sent %d bytes to syslog-ng", (int)sent);
489 // Notify network manager that Ethernet activity occurred
490 net_mgr_notify_eth_activity();
491 }
492
493 close(sock);
494 heap_caps_free(msg);
495}

497// Public: wait for IP on a given netif key ("WIFI_STA_DEF" or Ethernet key)
498// Wait for IP on a specific netif key (if netif_key==NULL, wait for any active netif).
499bool network_wait_for_ip(int max_retries, int delay_ms, const char *netif_key)
500{
501 int retries = 0;
502 esp_netif_ip_info_t ip;
503
504 while (retries < max_retries)
505 {
506 esp_netif_t *netif = NULL;
507
508 if (netif_key && netif_key[0] != '\0')
509 netif = esp_netif_get_handle_from_ifkey(netif_key);
510 else
511 netif = get_active_netif();
512
513 if (netif)
514 {
515 if (esp_netif_get_ip_info(netif, &ip) == ESP_OK)
516 {
517 if (ip.ip.addr != 0)
518 {
519 ESP_LOGI(TAG, "IP ready on %s: " IPSTR, netif_key ? netif_key : "active", IP2STR(&ip.ip));
520 return true;
521 }
522 }
523 }
524
525 ESP_LOGW(TAG, "Waiting for IP on %s... (%d/%d)",
526 netif_key ? netif_key : "active interface", retries + 1, max_retries);
527
528 vTaskDelay(pdMS_TO_TICKS(delay_ms));
529 retries++;
530 }
531
532 return false;
533}
