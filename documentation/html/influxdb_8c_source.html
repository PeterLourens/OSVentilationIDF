64static void escape_tag(char *dst, size_t dst_size, const char *src)
65{
66 size_t j = 0;
67 for (size_t i = 0; src[i] != '\0' && j + 2 < dst_size; i++)
68 {
69 char c = src[i];
70 if (c == ' ' || c == ',' || c == '=')
71 {
72 dst[j++] = '\\';
73 }
74 dst[j++] = c;
75 }
76 dst[j] = '\0';
77}

88bool influxdb_enqueue_sensor_line(int bus, int ch, float sensor_val[DATA_FIELDS], float avg_val[DATA_FIELDS])
89{
90 if (ch < 0)
91 return false;
92
93 // Small stack buffer per sensor; increase if you expect very long tag strings.
94 char line[512];
95 memset(line, 0, sizeof(line));
96
97 // Prepare tags (escaped)
98 char type_esc[64] = {0};
99 char valve_esc[64] = {0};
100 char loc_esc[128] = {0};
101
102 // You must provide a way to get meta for the sensor (same as in build_payload_sensors_for_bus)
103 const bus_sensor_t *meta = (bus == 0) ? &bus0_sensor_settings[ch] : &bus1_sensor_settings[ch];
104
105 escape_tag(type_esc, sizeof(type_esc), meta->type);
106 escape_tag(valve_esc, sizeof(valve_esc), meta->valve);
107 escape_tag(loc_esc, sizeof(loc_esc), meta->location);
108
109 if (type_esc[0] == '\0')
110 strcpy(type_esc, "None");
111 if (loc_esc[0] == '\0')
112 strcpy(loc_esc, "None");
113 if (valve_esc[0] == '\0')
114 strcpy(valve_esc, "None");
115
116 char sensor_id[32];
117 snprintf(sensor_id, sizeof(sensor_id), "bus%d_ch%d", bus, ch);
118
119 int64_t ts = influxdb_get_timestamp();
120
121 // Build the line. Keep format consistent with your existing code.
122 // If CO2 is present, append those fields.
123 int n = snprintf(line, sizeof(line),
124 "sensors,bus=%d,channel=%d,sensor_id=%s,type=%s,location=%s,valve=%s "
125 "temperature_raw=%.2f,temperature_avg=%.2f,humidity_raw=%.2f,humidity_avg=%.2f",
126 bus, ch, sensor_id, type_esc, loc_esc, valve_esc,
127 sensor_val[IDX_TEMP], avg_val[IDX_TEMP],
128 sensor_val[IDX_HUMIDITY], avg_val[IDX_HUMIDITY]);
129
130 if (n < 0 || n >= (int)sizeof(line))
131 {
132 ESP_LOGW(TAG, "sensor line truncated for bus=%d ch=%d (n=%d buf=%zu)", bus, ch, n, sizeof(line));
133 // If truncated, still attempt to enqueue what we have (or skip)
134 }
135 else
136 {
137 // Append CO2 fields if present
138 if (sensor_val[IDX_CO2] > 0.0f)
139 {
140 int m = snprintf(line + n, sizeof(line) - n, ",co2_raw=%.2f,co2_avg=%.2f", sensor_val[IDX_CO2], avg_val[IDX_CO2]);
141 if (m < 0 || m >= (int)(sizeof(line) - n))
142 {
143 ESP_LOGW(TAG, "sensor CO2 append truncated for bus=%d ch=%d", bus, ch);
144 }
145 else
146 {
147 n += m;
148 }
149 }
150
151 // Append timestamp and newline
152 int t = snprintf(line + n, sizeof(line) - n, " %lld\n", (long long)ts);
153 if (t < 0 || t >= (int)(sizeof(line) - n))
154 {
155 ESP_LOGW(TAG, "sensor timestamp append truncated for bus=%d ch=%d", bus, ch);
156 }
157 }
158
159 // Enqueue the single-line payload
160 if (strlen(line) == 0)
161 return false;
162 return influxdb_enqueue_payload(line, strlen(line));
163}

224// Get timestamp from DS3231 (UTC seconds) — ESP-IDF safe
230int64_t influxdb_get_timestamp(void)
231{
232 struct tm rtc_time = {0};
233
234 if (ds3231_get_time(ds3231_bus0_handle, &rtc_time) != ESP_OK)
235 {
236 time_t now;
237 time(&now);
238 return (int64_t)now * 1000000000LL;
239 }
240
241 int year = rtc_time.tm_year + 1900;
242 int month = rtc_time.tm_mon + 1;
243 int day = rtc_time.tm_mday;
244 int hour = rtc_time.tm_hour;
245 int min = rtc_time.tm_min;
246 int sec = rtc_time.tm_sec;
247
248 if (month <= 2)
249 {
250 year -= 1;
251 month += 12;
252 }
253
254 int64_t era = year / 400;
255 int64_t yoe = year - era * 400;
256 int64_t doy = (153 * (month - 3) + 2) / 5 + day - 1;
257 int64_t doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
258
259 int64_t days = era * 146097 + doe - 719468;
260 int64_t ts = days * 86400LL + hour * 3600 + min * 60 + sec;
261
262 return ts * 1000000000LL;
263}

302void influxdb_append_line(char *out, size_t out_size, const char *line)
303{
304 if (!out || out_size == 0 || !line)
305 return;
306
307 size_t used = strlen(out);
308 if (used >= out_size - 1)
309 return;
310
311 size_t out_free = out_size - used - 1;
312 size_t len = strlen(line);
313
314 if (len > out_free)
315 {
316 LOGE(TAG, "Influx payload overflow: line needs %zu bytes, free %zu", len, out_free);
317 len = out_free;
318 }
319
320 memcpy(out + used, line, len);
321 out[used + len] = '\0';
322}

370// Called from the InfluxDB task loop. Maintains an internal counter and enqueues DMA metrics
371// only when the configured multiplier condition is met. Returns true if a DMA metric was enqueued.
377bool influxdb_dma_monitor_tick(void)
378{
379 static uint32_t tick = 0;
380 tick++;
381
382 // Avoid division by zero
383 int mult = g_dma_monitor_multiplier > 0 ? g_dma_monitor_multiplier : 1;
384
385 if ((tick % (uint32_t)mult) == 0)
386 {
387 // Use a short host tag; replace with dynamic node id if you want (ensure thread-safety)
388 const char *host_tag = "osvent";
389 bool ok = influxdb_enqueue_dma_metrics(host_tag);
390
391 // Local debug log (keeps output small)
392 uint32_t dma_free = heap_caps_get_free_size(MALLOC_CAP_DMA);
393 uint32_t heap_free = esp_get_free_heap_size();
394 uint32_t largest = heap_caps_get_largest_free_block(MALLOC_CAP_INTERNAL);
395 ESP_LOGD("influxdb", "dma_monitor_tick: enqueued=%d dma_free=%u heap_free=%u largest=%u", ok ? 1 : 0, dma_free, heap_free, largest);
396
397 if (dma_free < 8000)
398 {
399 ESP_LOGW("influxdb", "dma_monitor_tick: DMA low: %u bytes", dma_free);
400 }
401
402 return ok;
403 }
404
405 return false;
406}

682// Low-level send (returns ESP_OK on HTTP 2xx)
690static esp_err_t influxdb_send_payload_internal(const char *payload, const Influxdb_config_t *cfg)
691{
692 if (!network_has_ip())
693 {
694 LOGW(TAG, "InfluxDB: Cannot send, no active network interface");
695 return ESP_ERR_INVALID_STATE;
696 }
697
698 if (!payload || !cfg)
699 {
700 LOGW(TAG, "send_internal: invalid args");
701 return ESP_ERR_INVALID_ARG;
702 }
703
704 char url[1024] = {0};
705 influxdb_build_url(url, sizeof(url), cfg);
706
707 esp_http_client_config_t http_cfg = {
708 .url = url,
709 .method = HTTP_METHOD_POST,
710 .timeout_ms = 15000,
711 };
712
713 esp_http_client_handle_t client = esp_http_client_init(&http_cfg);
714 if (!client)
715 {
716 LOGE(TAG, "send_internal: esp_http_client_init failed");
717 return ESP_FAIL;
718 }
719
720 char auth_header[600] = {0};
721 snprintf(auth_header, sizeof(auth_header), "Token %s", cfg->influxdb_token);
722
723 esp_http_client_set_header(client, "Authorization", auth_header);
724 esp_http_client_set_header(client, "Content-Type", "text/plain; charset=utf-8");
725 esp_http_client_set_header(client, "Accept", "application/json");
726
727 LOGD(TAG, "HTTP POST to: %s", url);
728 LOGD(TAG, "HTTP payload preview: %.256s", payload);
729
730 esp_err_t err = esp_http_client_set_post_field(client, payload, (int)strlen(payload));
731 if (err != ESP_OK)
732 {
733 LOGW(TAG, "esp_http_client_set_post_field failed: %s", esp_err_to_name(err));
734 esp_http_client_cleanup(client);
735 return err;
736 }
737
738 err = esp_http_client_perform(client);
739 int status = 0;
740 if (err == ESP_OK)
741 {
742 status = esp_http_client_get_status_code(client);
743 LOGD(TAG, "HTTP status: %d", status);
744
745 // Read response body safely
746 char resp[512];
747 int rlen = esp_http_client_read_response(client, resp, sizeof(resp) - 1);
748 if (rlen > 0)
749 {
750 resp[rlen] = '\0';
751 LOGI(TAG, "Influx response: %s", resp);
752 }
753 else
754 {
755 LOGD(TAG, "No response body (rlen=%d)", rlen);
756 }
757 }
758 else
759 {
760 LOGW(TAG, "HTTP perform error: %s", esp_err_to_name(err));
761 }
762
763 esp_http_client_cleanup(client);
764
765 if (err != ESP_OK)
766 return err;
767
768 if (status < 200 || status >= 300)
769 {
770 LOGW(TAG, "Influx write returned non-2xx status: %d", status);
771 return ESP_FAIL;
772 }
773
774 return ESP_OK;
775}

777// Send with retry/backoff (synchronous, called from sender task)
784static void influxdb_send_with_retry(const char *payload, const Influxdb_config_t *cfg)
785{
786 if (!payload || !cfg)
787 return;
788
789 const int max_retries = 4;
790 int attempt = 0;
791 int backoff_ms = 500;
792
793 while (attempt <= max_retries)
794 {
795 esp_err_t err = influxdb_send_payload_internal(payload, cfg);
796 if (err == ESP_OK)
797 {
798 LOGD(TAG, "Influx write succeeded (attempt %d)", attempt + 1);
799 return;
800 }
801
802 LOGW(TAG, "Influx write failed (attempt %d): %s", attempt + 1, esp_err_to_name(err));
803 vTaskDelay(pdMS_TO_TICKS(backoff_ms));
804 backoff_ms = backoff_ms * 2;
805 attempt++;
806 }
807
808 LOGE(TAG, "Influx write permanently failed after %d attempts", max_retries + 1);
809}

811// Sender task: serializes HTTP writes
817static void influx_sender_task(void *pv)
818{
819 Influxdb_config_t cfg;
820 influxdb_get_config_snapshot(&cfg);
821
822 while (!network_has_ip())
823 {
824 LOGW(TAG, "InfluxDB: No network available, waiting...");
825 vTaskDelay(pdMS_TO_TICKS(2000));
826 }
827
828 for (;;)
829 {
830 influx_job_t job;
831 if (xQueueReceive(influx_queue, &job, portMAX_DELAY) == pdTRUE)
832 {
833 if (!job.payload)
834 continue;
835
836 influxdb_get_config_snapshot(&cfg);
837
838 while (!network_has_ip())
839 {
840 LOGW(TAG, "InfluxDB: No network available, waiting...");
841 vTaskDelay(pdMS_TO_TICKS(2000));
842 }
843
844 if (strcmp(cfg.enable_influxdb, "On") != 0)
845 {
846 LOGD(TAG, "InfluxDB disabled; dropping payload");
847 heap_caps_free(job.payload);
848 continue;
849 }
850
851 influxdb_send_with_retry(job.payload, &cfg);
852 heap_caps_free(job.payload);
853 }
854 }
855}

863bool influxdb_init_sender(void)
864{
865 if (influx_queue != NULL)
866 {

891// Enqueue helper (non-blocking with short timeout)
899bool influxdb_enqueue_payload(const char *payload, size_t len)
900{
901 if (!payload || len == 0)
902 {
903 LOGW(TAG, "enqueue: empty payload");
904 return false;
905 }
906
907 if (!influx_queue)
908 {
909 LOGW(TAG, "enqueue: influx queue not initialized");
910 return false;
911 }
912
913 if (len > INFLUX_MAX_PAYLOAD)
914 {
915 LOGW(TAG, "enqueue: payload too large (%zu), truncating to %d", len, INFLUX_MAX_PAYLOAD);
916 len = INFLUX_MAX_PAYLOAD;
917 }
918
919 char *copy = heap_caps_malloc(len + 1, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
920 if (!copy)
921 {
922 LOGE(TAG, "enqueue: heap_caps_malloc failed for %zu bytes", len);
923 return false;
924 }
925
926 memcpy(copy, payload, len);
927 copy[len] = '\0';
928
929 influx_job_t job = {.payload = copy, .len = len};
930
931 if (xQueueSend(influx_queue, &job, pdMS_TO_TICKS(1000)) != pdTRUE)
932 {
933 LOGW(TAG, "enqueue: queue full after waiting 1s, dropping payload len=%zu", len);
934 heap_caps_free(copy);
935 return false;
936 }
937
938 LOGD(TAG, "Enqueued Influx payload len=%zu", len);
939 return true;
940}

942// High-level process function (reworked to enqueue separate payloads)
949void influxdb_process(QueueHandle_t sensor_q, QueueHandle_t avg_q)
950{
951 Influxdb_config_t cfg;
952 influxdb_get_config_snapshot(&cfg);
953
954 if (strcmp(cfg.enable_influxdb, "On") != 0)
955 return;
956
957 // Build and enqueue uptime-only actuals first, to keep uptime updating even when full state is delayed.
958 char *uptime_payload = heap_caps_malloc(256, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
959 if (uptime_payload)
960 {
961 memset(uptime_payload, 0, 256);
962 influxdb_build_payload_uptime(uptime_payload, 256);
963 if (strlen(uptime_payload) > 0)
964 {
965 if (!influxdb_enqueue_payload(uptime_payload, strlen(uptime_payload)))
966 {
967 LOGW(TAG, "influxdb_process: failed to enqueue uptime payload");
968 }
969 }
970 heap_caps_free(uptime_payload);
971 }
972
973 // Build and enqueue actuals
974 char *actuals = heap_caps_malloc(2048, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
975 if (actuals)
976 {
977 memset(actuals, 0, 2048);
978 influxdb_build_payload_actuals(actuals, 2048);
979 if (strlen(actuals) > 0)
980 {
981 if (!influxdb_enqueue_payload(actuals, strlen(actuals)))
982 {
983 LOGW(TAG, "influxdb_process: failed to enqueue actuals payload");
984 }
985 }
986 else
987 {
988 LOGW(TAG, "influxdb_process: actuals payload empty, skipping actuals write");
989 }
990 heap_caps_free(actuals);
991 }
992
993 // Build and enqueue valves
994 char *valves = heap_caps_malloc(2048, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
995 if (valves)
996 {
997 memset(valves, 0, 2048);
998 influxdb_build_payload_valves(valves, 2048);
999 if (strlen(valves) > 0)
1000 influxdb_enqueue_payload(valves, strlen(valves));
1001 heap_caps_free(valves);
1002 }
1003
1004 // Allocate sensor buffers
1005 size_t buf_bytes = sizeof(float) * MAX_I2C_BUSES * CHANNELS_PER_BUS * DATA_FIELDS;
1006 float (*sensor)[CHANNELS_PER_BUS][DATA_FIELDS] = heap_caps_malloc(buf_bytes, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
1007 float (*avg)[CHANNELS_PER_BUS][DATA_FIELDS] = heap_caps_malloc(buf_bytes, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
1008
1009 if (!sensor || !avg)
1010 {
1011 LOGE(TAG, "Memory allocation failed for sensors");
1012 heap_caps_free(sensor);
1013 heap_caps_free(avg);
1014 return;
1015 }
1016
1017 memset(sensor, 0, buf_bytes);
1018 memset(avg, 0, buf_bytes);
1019
1020 if (!sensor_q || !avg_q)
1021 {
1022 LOGW(TAG, "influxdb_process: sensor queues not provided");
1023 heap_caps_free(sensor);
1024 heap_caps_free(avg);
1025 return;
1026 }
1027
1028 if (xQueuePeek(sensor_q, sensor, 0) != pdTRUE || xQueuePeek(avg_q, avg, 0) != pdTRUE)
1029 {
1030 LOGW(TAG, "Failed to read sensor queues");
1031 heap_caps_free(sensor);
1032 heap_caps_free(avg);
1033 return;
1034 }
1035
1036 // Send each sensor as a separate Influx line to avoid large allocations
1037 for (int bus = 0; bus < MAX_I2C_BUSES; ++bus)
1038 {
1039 for (int ch = 0; ch < CHANNELS_PER_BUS; ++ch)
1040 {
1041 // Skip invalid sensors (same check as before)
1042 if (sensor[bus][ch][IDX_VALID] != 1.0f)
1043 continue;
1044
1045 // Build and enqueue one sensor line
1046 bool ok = influxdb_enqueue_sensor_line(bus, ch, sensor[bus][ch], avg[bus][ch]);
1047 if (!ok)
1048 {
1049 ESP_LOGW(TAG, "Failed to enqueue sensor line bus=%d ch=%d", bus, ch);
1050 }
1051 }
1052 }
1053
1054 heap_caps_free(sensor);
1055 heap_caps_free(avg);
1056}
