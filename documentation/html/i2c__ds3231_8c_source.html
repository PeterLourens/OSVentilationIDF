19esp_err_t i2c_ds3231_init(i2c_master_bus_handle_t bus_handle, i2c_master_dev_handle_t *dev_handle)
20{
21 i2c_device_config_t ds3231_cfg = {
22 .dev_addr_length = I2C_ADDR_BIT_LEN_7,
23 .device_address = DS3231_I2C_ADDR, // 0x68
24 .scl_speed_hz = 100000,
25 };
26
27 esp_err_t err = i2c_master_bus_add_device(bus_handle, &ds3231_cfg, dev_handle);
28 if (err != ESP_OK)
29 {
30 LOGE(TAG, "Failed to add DS3231 device: %s", esp_err_to_name(err));
31 return err;
32 }
33
34 // Check if device responds by reading status register (0x0F)
35 uint8_t reg = 0x0F;
36 uint8_t status = 0;
37
38 err = i2c_master_transmit_receive(*dev_handle, &reg, 1, &status, 1, pdMS_TO_TICKS(100));
39 if (err != ESP_OK)
40 {
41 LOGE(TAG, "Failed to read status register: %s", esp_err_to_name(err));
42 return err;
43 }
44
45 LOGI(TAG, "DS3231 initialized successfully, status=0x%02X", status);
46 return ESP_OK;
47}

52void init_system_time_from_ds3231(i2c_master_dev_handle_t dev_handle)
53{
54 struct tm rtc_time;
55 time_t sys_now;
56 struct tm sys_time;
57
58 if (ds3231_get_time(dev_handle, &rtc_time) == ESP_OK)
59 {
60 time_t rtc_epoch = mktime(&rtc_time);
61 struct timeval now = {
62 .tv_sec = rtc_epoch,
63 .tv_usec = 0,
64 };
65
66 settimeofday(&now, NULL);
67
68 // Print both RTC and system time
69 time(&sys_now);
70 localtime_r(&sys_now, &sys_time);
71
72 LOGD(TAG, "RTC time read: %s", asctime(&rtc_time));
73 LOGD(TAG, "System time set: %s", asctime(&sys_time));
74 }
75 else
76 {
77 LOGE(TAG, "Failed to read DS3231 at boot, system time starts at default (1970)");
78 }
79}

100esp_err_t ds3231_set_time(i2c_master_dev_handle_t dev, const struct tm *timeinfo)
101{
102 uint8_t buf[8];
103 buf[0] = 0x00; // start writing at register 0x00 (seconds)
104
105 buf[1] = dec_to_bcd(timeinfo->tm_sec);
106 buf[2] = dec_to_bcd(timeinfo->tm_min);
107 buf[3] = dec_to_bcd(timeinfo->tm_hour);
108 buf[4] = dec_to_bcd(timeinfo->tm_wday ? timeinfo->tm_wday : 7);
109 buf[5] = dec_to_bcd(timeinfo->tm_mday);
110 buf[6] = dec_to_bcd(timeinfo->tm_mon + 1);
111 buf[7] = dec_to_bcd(timeinfo->tm_year % 100); // 2000–2099
112
113 return i2c_master_transmit(dev, buf, sizeof(buf), -1);
114}

119esp_err_t ds3231_get_time(i2c_master_dev_handle_t dev, struct tm *timeinfo)
120{
121 uint8_t reg = 0x00;
122 uint8_t buf[7];
123 esp_err_t err = i2c_master_transmit_receive(dev, &reg, 1, buf, sizeof(buf), -1);
124 if (err != ESP_OK)
125 return err;
126
127 timeinfo->tm_sec = bcd_to_dec(buf[0] & 0x7F);
128 timeinfo->tm_min = bcd_to_dec(buf[1] & 0x7F);
129 timeinfo->tm_hour = bcd_to_dec(buf[2] & 0x3F);
130 timeinfo->tm_wday = bcd_to_dec(buf[3] & 0x07);
131 timeinfo->tm_mday = bcd_to_dec(buf[4] & 0x3F);
132 timeinfo->tm_mon = bcd_to_dec(buf[5] & 0x1F) - 1;
133
134 uint8_t year_bcd = buf[6] & 0x7F;
135 uint8_t century = (buf[6] & 0x80) ? 1 : 0;
136
137 int year = bcd_to_dec(year_bcd); // 0–99
138
139 if (century == 0)
140 year += 2000;
141 else
142 year += 2100;
143
144 timeinfo->tm_year = year - 1900; // struct tm format
145
146 return ESP_OK;
147}

152esp_err_t sync_system_and_ds3231_with_ntp(i2c_master_dev_handle_t dev_handle)
153{
154 char ntp_server[64];
155 char timezone[64];
156
157 // Read config under mutex
158 if (xSemaphoreTake(time_config_mutex, pdMS_TO_TICKS(20)) != pdTRUE)
159 {
160 LOGE(TAG, "Failed to acquire time_config mutex");
161 return ESP_ERR_TIMEOUT;
162 }
163
164 strncpy(ntp_server, time_config.ntp_server, sizeof(ntp_server));
165 ntp_server[sizeof(ntp_server) - 1] = '\0';
166 strncpy(timezone, time_config.timezone, sizeof(timezone));
167 timezone[sizeof(timezone) - 1] = '\0';
168 xSemaphoreGive(time_config_mutex);
169
170 // Check network availability (Ethernet preferred, Wi-Fi fallback)
171 if (!network_has_ip())
172 {
173 LOGW(TAG, "No network available, fallback to DS3231");
174
175 struct tm rtc_time;
176 if (ds3231_get_time(dev_handle, &rtc_time) == ESP_OK)
177 {
178 struct timeval tv = {.tv_sec = mktime(&rtc_time), .tv_usec = 0};
179 settimeofday(&tv, NULL);
180 LOGI(TAG, "System time set from DS3231");
181 return ESP_OK;
182 }
183
184 LOGE(TAG, "Failed to read DS3231");
185 return ESP_FAIL;
186 }
187
188 // Configure timezone
189 // Set at start, not here
190 // setenv("TZ", timezone, 1);
191 // tzset();
192
193 // Configure SNTP
194 esp_sntp_stop();
195 esp_sntp_setoperatingmode(ESP_SNTP_OPMODE_POLL);
196 esp_sntp_setservername(0, ntp_server);
197 esp_sntp_init();
198
199 LOGI(TAG, "SNTP started with server: %s, TZ: %s", ntp_server, timezone);
200
201 // Wait for sync
202 int retry = 0;
203 const int retry_count = 10;
204 while (retry < retry_count)
205 {
206 time_t now = time(NULL);
207 if (now > 1600000000)
208 break; // sanity check: year > 2020
209 LOGI(TAG, "Waiting for NTP time... (%d/%d)", retry + 1, retry_count);
210 vTaskDelay(pdMS_TO_TICKS(2000));
211 retry++;
212 }
213
214 if (time(NULL) < 1600000000)
215 {
216 LOGE(TAG, "NTP sync failed");
217 return ESP_FAIL;
218 }
219
220 // Log current time
221 time_t now;
222 struct tm timeinfo;
223 time(&now);
224
225 // Write UTC to DS3231, not localtime
226 // localtime_r(&now, &timeinfo);
227 gmtime_r(&now, &timeinfo);
228
229 char buf[64];
230 strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &timeinfo);
231 LOGI(TAG, "System time synchronized (UTC): %s", buf);
232
233 // Update DS3231
234 esp_err_t err = ds3231_set_time(dev_handle, &timeinfo);
235 if (err == ESP_OK)
236 {
237 LOGI(TAG, "DS3231 updated successfully");
238 }
239 else
240 {
241 LOGE(TAG, "Failed to update DS3231: %s", esp_err_to_name(err));
242 return err;
243 }
244
245 return ESP_OK;
246}

266// Temperature
270esp_err_t ds3231_read_temperature(i2c_master_dev_handle_t dev, float *temp_c)
271{
272 uint8_t reg = DS3231_REG_TEMP_MSB;
273 uint8_t buf[2];
274 esp_err_t err = i2c_master_transmit_receive(dev, &reg, 1, buf, 2, pdMS_TO_TICKS(100));
275 if (err != ESP_OK)
276 return err;
277
278 int8_t msb = buf[0]; // signed
279 uint8_t lsb = buf[1] >> 6; // upper 2 bits
280 *temp_c = msb + (lsb * 0.25f);
281 return ESP_OK;
282}

284// Status + health check
288esp_err_t ds3231_check_status(i2c_master_dev_handle_t dev, char *buf, size_t bufsize)
289{
290 uint8_t status;
291 esp_err_t err = ds3231_read_reg(dev, DS3231_REG_STATUS, &status);
292 if (err != ESP_OK)
293 {
294 LOGE(TAG, "Failed to read status: %s", esp_err_to_name(err));
295 return err;
296 }
297
298 bool osf = status & 0x80; // Oscillator Stop Flag
299 bool busy = status & 0x04; // Busy flag (temp conversion)
300 bool alarm1 = status & 0x01;
301 bool alarm2 = status & 0x02;
302
303 if (osf)
304 {
305 LOGW(TAG, "Oscillator Stop Flag set — RTC may have lost power!");
306 ds3231_write_reg(dev, DS3231_REG_STATUS, status & ~0x80); // Clear OSF
307 }
308 else
309 {
310 LOGD(TAG, "Oscillator running normally");
311 }
312
313 snprintf(buf, bufsize, "OSF=%d Busy=%d\nAl1=%d Al2=%d", osf, busy, alarm1, alarm2);
314
315 LOGD(TAG, "Status bits: OSF=%d, Busy=%d, Alarm1=%d, Alarm2=%d", osf, busy, alarm1, alarm2);
316
317 return ESP_OK;
318}

339void save_ds3231_diagnostics(float temp_c)
340{
341 nvs_handle_t handle;
342
343 esp_err_t err = nvs_open(TAG, NVS_READWRITE, &handle);
344 if (err != ESP_OK)
345 {
346 LOGE(TAG, "Error opening NVS: %s", esp_err_to_name(err));
347 return;
348 }
349
350 nvs_set_i32(handle, "temp_c", (int32_t)(temp_c * 100));
351
352 // Store current epoch time
353 time_t now;
354 time(&now);
355 nvs_set_i64(handle, "last_check", (int64_t)now);
356
357 nvs_commit(handle);
358 nvs_close(handle);
359
360 LOGD(TAG, "Diagnostics saved: temp=%.2f °C", temp_c);
361}

366void read_ds3231_diagnostics(void)
367{
368 nvs_handle_t handle;
369 esp_err_t err = nvs_open(TAG, NVS_READONLY, &handle);
370
371 if (err != ESP_OK)
372 {
373 LOGW(TAG, "No diagnostics stored yet");
374 return;
375 }
376
377 int32_t temp100;
378 int64_t last_check;
379
380 if (nvs_get_i32(handle, "temp_c", &temp100) == ESP_OK && nvs_get_i64(handle, "last_check", &last_check) == ESP_OK)
381 {
382 struct tm tm;
383
384 localtime_r((time_t *)&last_check, &tm);
385 LOGD(TAG, "Last temp: %.2f °C", temp100 / 100.0f);
386 LOGD(TAG, "Last check: %s", asctime(&tm));
387 }
388 else
389 {
390 LOGW(TAG, "Diagnostics keys not found");
391 }
392
393 nvs_close(handle);
394}
