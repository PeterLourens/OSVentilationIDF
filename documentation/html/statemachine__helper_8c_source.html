16// Find state index by name (case-insensitive). Returns -1 if not found.
17int find_state_index_by_name_ci(const char *name)
18{

20 return -1;
21
22 for (int i = 0; i < MAX_NR_STATES; ++i)
23 {
24 const char *s = statemachine_config[i].state_name;
25 if (!s)
26 continue;
27 if (strcasecmp(s, name) == 0)
28 return i;
29 }
30 return -1;
31}

35{

37 int wkend_d_h = 0, wkend_d_m = 0, wkend_n_h = 0, wkend_n_m = 0;

39 if (state_config_mutex && xSemaphoreTake(state_config_mutex, pdMS_TO_TICKS(50)) == pdTRUE)
40 {
41 wday_d_h = statemachine_gen_config.wday_d_h_start;
42 wday_d_m = statemachine_gen_config.wday_d_m_start;
43 wday_n_h = statemachine_gen_config.wday_n_h_start;
44 wday_n_m = statemachine_gen_config.wday_n_m_start;
45 wkend_d_h = statemachine_gen_config.wkend_d_h_start;
46 wkend_d_m = statemachine_gen_config.wkend_d_m_start;
47 wkend_n_h = statemachine_gen_config.wkend_n_h_start;
48 wkend_n_m = statemachine_gen_config.wkend_n_m_start;
49 xSemaphoreGive(state_config_mutex);
50 }
51 else
52 {
53 LOGW(TAG, "is_day: state_config_mutex unavailable, using defaults");
54 }
55
56 time_t now = time(NULL);
57 struct tm tm_now;
58 localtime_r(&now, &tm_now);
59 int current_minutes = tm_now.tm_hour * 60 + tm_now.tm_min;
60
61 if (is_weekend())
62 {
63 int day_start = wkend_d_h * 60 + wkend_d_m;
64 int night_start = wkend_n_h * 60 + wkend_n_m;
65 // If start == stop, treat as full-day
66 if (day_start == night_start)
67 return true;
68 if (day_start < night_start)
69 return (current_minutes >= day_start && current_minutes < night_start);
70 // wrapping window
71 return (current_minutes >= day_start || current_minutes < night_start);
72 }
73 else
74 {
75 int day_start = wday_d_h * 60 + wday_d_m;
76 int night_start = wday_n_h * 60 + wday_n_m;
77 if (day_start == night_start)
78 return true;
79 if (day_start < night_start)
80 return (current_minutes >= day_start && current_minutes < night_start);
81 return (current_minutes >= day_start || current_minutes < night_start);
82 }

158// Helper: compute overlap minutes between two windows [a_start,a_stop) and [b_start,b_stop)
159static int minutes_overlap(int a_start, int a_stop, int b_start, int b_stop)
160{
161 int total = 0;
162
163 int a_s[2], a_e[2], a_cnt = 0;
164 if (a_start == a_stop)
165 {
166 a_s[0] = 0;
167 a_e[0] = 1440;
168 a_cnt = 1;
169 }
170 else if (a_start < a_stop)
171 {
172 a_s[0] = a_start;
173 a_e[0] = a_stop;
174 a_cnt = 1;
175 }
176 else
177 {
178 a_s[0] = a_start;
179 a_e[0] = 1440;
180 a_s[1] = 0;
181 a_e[1] = a_stop;
182 a_cnt = 2;
183 }
184
185 int b_s[2], b_e[2], b_cnt = 0;
186 if (b_start == b_stop)
187 {
188 b_s[0] = 0;
189 b_e[0] = 1440;
190 b_cnt = 1;
191 }
192 else if (b_start < b_stop)
193 {
194 b_s[0] = b_start;
195 b_e[0] = b_stop;
196 b_cnt = 1;
197 }
198 else
199 {
200 b_s[0] = b_start;
201 b_e[0] = 1440;
202 b_s[1] = 0;
203 b_e[1] = b_stop;
204 b_cnt = 2;
205 }
206
207 for (int i = 0; i < a_cnt; ++i)
208 {
209 for (int j = 0; j < b_cnt; ++j)
210 {
211 int ov_s = (a_s[i] > b_s[j]) ? a_s[i] : b_s[j];
212 int ov_e = (a_e[i] < b_e[j]) ? a_e[i] : b_e[j];
213 if (ov_e > ov_s)
214 total += ov_e - ov_s;
215 }
216 }
217
218 return total;
219}

221// Determine whether cycling should run during the day window
222bool valve_cycle_times_day(void)
223{
224 int idx = find_state_index_by_name_ci(STATE_CYCLINGDAY);
225 if (idx < 0)
226 {
227 LOGE(TAG, "valve_cycle_times_day: state not configured");
228 return false;
229 }
230
231 int start_h, start_m, stop_h, stop_m, interval_min, duration_min;
232 int wday_d_h, wday_d_m, wday_n_h, wday_n_m;
233 int wkend_d_h, wkend_d_m, wkend_n_h, wkend_n_m;
234
235 if (state_config_mutex && xSemaphoreTake(state_config_mutex, pdMS_TO_TICKS(100)) == pdTRUE)
236 {
237 start_h = statemachine_config[idx].state_start_hour;
238 start_m = statemachine_config[idx].state_start_minute;
239 stop_h = statemachine_config[idx].state_end_hour;
240 stop_m = statemachine_config[idx].state_end_minute;
241 interval_min = statemachine_config[idx].cycling_interval_min;
242 duration_min = statemachine_config[idx].duration_minutes;
243
244 wday_d_h = statemachine_gen_config.wday_d_h_start;
245 wday_d_m = statemachine_gen_config.wday_d_m_start;
246 wday_n_h = statemachine_gen_config.wday_n_h_start;
247 wday_n_m = statemachine_gen_config.wday_n_m_start;
248
249 wkend_d_h = statemachine_gen_config.wkend_d_h_start;
250 wkend_d_m = statemachine_gen_config.wkend_d_m_start;
251 wkend_n_h = statemachine_gen_config.wkend_n_h_start;
252 wkend_n_m = statemachine_gen_config.wkend_n_m_start;
253
254 xSemaphoreGive(state_config_mutex);
255 }
256 else
257 {
258 LOGE(TAG, "valve_cycle_times_day: could not take state_config_mutex");
259 return false;
260 }
261
262 if (interval_min < 30 || duration_min <= 5)
263 {
264 LOGD(TAG, "valve_cycle_times_day: invalid interval/duration interval=%d duration=%d", interval_min, duration_min);
265 return false;
266 }
267
268 time_t now = time(NULL);
269 struct tm tm_now;
270 localtime_r(&now, &tm_now);
271 int current = tm_now.tm_hour * 60 + tm_now.tm_min;
272 int start = start_h * 60 + start_m;
273 int stop = stop_h * 60 + stop_m;
274
275 // Check if current is inside the configured cycling window
276 bool in_window;
277 if (start == stop)
278 in_window = true;
279 else if (start < stop)
280 in_window = (current >= start && current < stop);
281 else
282 in_window = (current >= start || current < stop);
283
284 if (!in_window)
285 return false;
286
287 int day_start, day_stop;
288 if (is_weekend())
289 {
290 day_start = wkend_d_h * 60 + wkend_d_m;
291 day_stop = wkend_n_h * 60 + wkend_n_m;
292 }
293 else
294 {
295 day_start = wday_d_h * 60 + wday_d_m;
296 day_stop = wday_n_h * 60 + wday_n_m;
297 }
298
299 int overlap = minutes_overlap(start, stop, day_start, day_stop);
300 if (overlap <= 0)
301 return false;
302
303 int max_cycles = ((overlap - 1) / interval_min) + 1;
304 if (max_cycles <= 0)
305 return false;
306
307 int elapsed = (current >= start) ? (current - start) : (current + 1440 - start);
308 int cycle_index = elapsed / interval_min;
309 if (cycle_index >= max_cycles)
310 return false;
311
312 int elapsed_mod = elapsed % interval_min;
313 return (elapsed_mod < duration_min);
314}

416// Select sensors for CO2 and RH lists using averaged data
417void select_sensors(void)
418{
419 int j = 0; // CO2 counter
420 int k = 0; // RH counter
421
422 co2_sensor_counter = 0;
423 rh_sensor_counter = 0;
424
425 // Ensure queue is ready and item size matches
426 if (!sensors_queues_ready || sensor_avg_queue == NULL)
427 {
428 LOGW(TAG, "select_sensors: sensor_avg_queue not ready");
429 return;
430 }
431 if (sensor_avg_item_size != sizeof(float[MAX_I2C_BUSES][CHANNELS_PER_BUS][DATA_FIELDS]))
432 {
433 LOGW(TAG, "select_sensors: sensor_avg_item_size mismatch");
434 return;
435 }
436
437 float sensor_data[MAX_I2C_BUSES][CHANNELS_PER_BUS][DATA_FIELDS];
438 memset(sensor_data, 0, sizeof(sensor_data));
439
440 if (xQueuePeek(sensor_avg_queue, sensor_data, pdMS_TO_TICKS(50)) != pdTRUE)
441 {
442 LOGW(TAG, "select_sensors: failed to peek sensor_avg_queue");
443 return;
444 }
445
446 for (int i = 0; i < CHANNELS_PER_BUS; ++i)
447 {
448 const bus_sensor_t *s0 = NULL;
449 const bus_sensor_t *s1 = NULL;

451 if (sensor_cfg_mutex && xSemaphoreTake(sensor_cfg_mutex, pdMS_TO_TICKS(50)) == pdTRUE)

454 s1 = &bus1_sensor_settings[i];
455 xSemaphoreGive(sensor_cfg_mutex);
456 }
457 else

461
462 // Bus 0 CO2
